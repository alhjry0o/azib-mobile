```
// Pure routing decisions for the shell. No React, fully unit-testable.
import type { AuthSession, AuthUser } from '@/providers';
import type { Result } from '@/core/result';

export type GateStatus =
  | 'loading'
  | 'onboarding'
  | 'auth'
  | 'ready'
  | 'not-connected'
  | 'error';

export interface GateInput {
  /** Both persisted stores (app-mode + onboarding) finished hydrating. */
  readonly hydrated: boolean;
  readonly onboardingCompleted: boolean;
  readonly sessionLoading: boolean;
  readonly session: Result<AuthSession | null> | undefined;
}

export function resolveGate(input: GateInput): GateStatus {
  if (!input.hydrated || input.sessionLoading || input.session === undefined) {
    return 'loading';
  }
  if (!input.onboardingCompleted) return 'onboarding';
  if (!input.session.ok) {
    return input.session.error.kind === 'NotConnected' ? 'not-connected' : 'error';
  }
  return input.session.value !== null ? 'ready' : 'auth';
}

/** Removes Expo Router group segments: '/(tabs)/services' -> '/services'. */
export function stripGroups(path: string): string {
  const segments = path
    .split('/')
    .filter((s) => s.length > 0 && !(s.startsWith('(') && s.endsWith(')')));
  return `/${segments.join('/')}`;
}

export function isTabActive(pathname: string, tabPath: string): boolean {
  const current = stripGroups(pathname);
  if (tabPath === '/') return current === '/' || current === '/index';
  return current === tabPath;
}

/**
 * Staff role check. AuthUser has no roles field until P06/P17 deliver it.
 * Deny-by-default: nobody passes until roles exist. The UI guard is a
 * convenience only; real enforcement is server-side RLS (P18), per rule #19.
 */
export function hasStaffRole(_user: AuthUser): boolean {
  return false;
}
```

```
// Session access via AuthProvider.getSession() (P03 contract), cached by Query.
// Components never call providers directly elsewhere; they use this hook.
import { useCallback } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import type { AuthSession } from '@/providers';
import { useProviderContainer } from '@/providers';
import type { Result } from '@/core/result';
import type { AppMode } from '@/stores';

export function sessionQueryKey(mode: AppMode) {
  return ['auth', 'session', mode] as const;
}

export interface SessionApi {
  readonly result: Result<AuthSession | null> | undefined;
  readonly isLoading: boolean;
  refetch(): void;
  setSession(session: AuthSession | null): void;
}

export function useSession(): SessionApi {
  const container = useProviderContainer();
  const client = useQueryClient();
  const mode = container.mode;
  const key = sessionQueryKey(mode);

  const query = useQuery({
    queryKey: key,
    queryFn: () => container.auth.getSession(),
    staleTime: 30_000,
  });

  const setSession = useCallback(
    (session: AuthSession | null) => {
      client.setQueryData(sessionQueryKey(mode), { ok: true, value: session });
    },
    [client, mode],
  );

  const refetch = useCallback(() => {
    void query.refetch();
  }, [query]);

  return { result: query.data, isLoading: query.isLoading, refetch, setSession };
}
```

```
// Combines persisted stores + session query into one gate decision.
import { useAppModeStore, useOnboardingStore } from '@/stores';
import { resolveGate, type GateStatus } from './routing';
import { useSession } from './use-session';

export interface ShellGate {
  readonly status: GateStatus;
  refetch(): void;
}

export function useShellGate(): ShellGate {
  const modeHydrated = useAppModeStore((s) => s.hydrated);
  const onboardingHydrated = useOnboardingStore((s) => s.hydrated);
  const completed = useOnboardingStore((s) => s.completed);
  const session = useSession();

  const status = resolveGate({
    hydrated: modeHydrated && onboardingHydrated,
    onboardingCompleted: completed,
    sessionLoading: session.isLoading,
    session: session.result,
  });

  return { status, refetch: session.refetch };
}
```

```
// Deep-link allowlist for the azib:// scheme (P05).
// - allow:    public destinations; the router handles them normally.
// - guarded:  protected areas; their layout guards decide (staff/dev/onboarding).
// - not-found: everything else -> +not-found.
// Query strings and fragments are dropped entirely: no sensitive data in links.
import { stripGroups } from './routing';

export type DeepLinkResolution =
  | { readonly kind: 'allow'; readonly path: string }
  | { readonly kind: 'guarded'; readonly path: string }
  | { readonly kind: 'not-found'; readonly path: string }
  | { readonly kind: 'ignore' };

const ALLOWED: ReadonlySet<string> = new Set([
  '/',
  '/index',
  '/sign-in',
  '/maintenance',
  '/services',
  '/weddings',
  '/partners',
  '/account',
  '/assistance',
  '/souk',
  '/jobs',
  '/volunteers',
  '/learn',
  '/calculator',
  '/guide',
  '/notifications',
  '/search',
]);

const GUARDED_PREFIXES: readonly string[] = ['/onboarding', '/staff', '/dev', '/gallery'];

/** Parses azib:// URLs into a normalized, group-free path. Non-azib URLs -> null. */
export function parseAzibUrl(url: string): string | null {
  const match = /^azib:\/\/(.*)$/i.exec(url.trim());
  if (!match) return null;
  const rest = (match[1] ?? '').split(/[?#]/)[0] ?? '';
  const withSlash = rest.startsWith('/') ? rest : `/${rest}`;
  const stripped = stripGroups(withSlash);
  if (stripped.length > 1 && stripped.endsWith('/')) {
    return stripped.slice(0, -1);
  }
  return stripped;
}

export function resolveDeepLink(url: string | null | undefined): DeepLinkResolution {
  if (!url) return { kind: 'ignore' };
  const path = parseAzibUrl(url);
  if (path === null) return { kind: 'ignore' };
  if (ALLOWED.has(path)) return { kind: 'allow', path };
  const guarded = GUARDED_PREFIXES.some(
    (prefix) => path === prefix || path.startsWith(`${prefix}/`),
  );
  if (guarded) return { kind: 'guarded', path };
  return { kind: 'not-found', path };
}
```

```
export interface ReadOnlyRepository<T> {
  getById(id: string): Promise<Result<T | null>>;
  list(query?: ListQuery): Promise<Result<PagedResult<T>>>;
}

export interface AppendOnlyRepository<T, TCreate> {
  getById(id: string): Promise<Result<T | null>>;
  create(input: TCreate): Promise<Result<T>>;

}
```

```
  CreateProfileInput,

  ProfileListQuery,

  readonly forceErrorKind?: 'Network' | 'Unknown' | 'Validation' | null;
}

export function createMockProfileRepository(
  options: MockRepositoryOptions = {},
): ProfileRepository {
  const { delayMs = 0, forceErrorKind = null } = options;
  const byId = new Map<string, Profile>();

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    return null;
  }

  function nowIso(): string {
    return new Date().toISOString();
  }

  return Object.freeze({
    async getById(id: string): Promise<Result<Profile | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      return ok(byId.get(id) ?? null);
    },

    async getByUserId(userId: string): Promise<Result<Profile | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      for (const p of byId.values()) if (p.userId === userId) return ok(p);
      return ok(null);
    },

    async create(input: CreateProfileInput): Promise<Result<Profile>> {
      const parsed = CreateProfileInputSchema.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const id = `mock-profile-${Date.now()}-${byId.size}`;
      const profile: Profile = Object.freeze({
        id,
        userId: parsed.data.userId,
        displayName: parsed.data.displayName,
        gender: parsed.data.gender,
        maritalStatus: parsed.data.maritalStatus,
        city: parsed.data.city ?? null,
        phone: parsed.data.phone ?? null,
        createdAt: nowIso(),
        updatedAt: nowIso(),
        dataOrigin: 'demo' as const,
      });
      byId.set(id, profile);
      return ok(profile);
    },

    async update(id: string, input: UpdateProfileInput): Promise<Result<Profile>> {
      const parsed = UpdateProfileInputSchema.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const existing = byId.get(id);
      if (!existing) {
        return { ok: false, error: AppError.notFound({ code: 'PROFILE_NOT_FOUND' }) };
      }
      const updated: Profile = Object.freeze({
        ...existing,
        ...(parsed.data.displayName !== undefined && { displayName: parsed.data.displayName }),
        ...(parsed.data.gender !== undefined && { gender: parsed.data.gender }),
        ...(parsed.data.maritalStatus !== undefined && {
          maritalStatus: parsed.data.maritalStatus,
        }),
        ...(parsed.data.city !== undefined && { city: parsed.data.city }),
        ...(parsed.data.phone !== undefined && { phone: parsed.data.phone }),
        updatedAt: nowIso(),
      });
      byId.set(id, updated);
      return ok(updated);
    },

    async delete(id: string): Promise<Result<void>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      byId.delete(id);
      return ok(undefined);
    },

      if (f) return { ok: false, error: f };
      const limit = query?.limit ?? 20;
```

```
// Shared configuration for mock adapters.
// Tests can set delayMs > 0 to simulate latency, or forceError to test failure paths.

import { AppError } from '@/core/errors';
import type { AppErrorKind } from '@/core/errors';

export interface MockProviderConfig {
  /** Simulated latency in ms. Default 0 (instant). */
  readonly delayMs: number;
  /** When set, every call fails with this AppError kind. Default null. */
  readonly forceError: AppErrorKind | null;
}

export const DEFAULT_MOCK_CONFIG: MockProviderConfig = Object.freeze({
  delayMs: 0,
  forceError: null,
});

export function withConfig(partial: Partial<MockProviderConfig> = {}): MockProviderConfig {
  return Object.freeze({
    delayMs: partial.delayMs ?? DEFAULT_MOCK_CONFIG.delayMs,
    forceError: partial.forceError ?? DEFAULT_MOCK_CONFIG.forceError,
  });
}

/**
 * Returns an AppError to fail the call when forceError is set.
 * Otherwise awaits delayMs (if > 0) and returns null.
 */
export async function simulateMockIO(config: MockProviderConfig): Promise<AppError | null> {
  if (config.delayMs > 0) {
    await new Promise<void>((resolve) => setTimeout(resolve, config.delayMs));
  }
  if (config.forceError !== null) {
    return AppError.fromKind(config.forceError, { code: 'MOCK_FORCED_ERROR' });
  }
  return null;
}

/** Deterministic 32-bit hash for stable demo ids. */
export function stableHash(input: string): string {
  let h = 0;
  for (let i = 0; i < input.length; i += 1) {
    h = (h * 31 + input.charCodeAt(i)) | 0;
  }
  return Math.abs(h).toString(36);
}
```

```
// In-memory FileStorageProvider. Files are opaque URIs; no actual I/O.

import { AppError } from '@/core/errors';
import type { Result } from '@/core/result';

import type { FileStorageProvider, FileUploadOptions, UploadedFile } from '../contracts/storage';
import type { HealthCheckResult } from '../types';
import { DEFAULT_MOCK_CONFIG, simulateMockIO, stableHash, type MockProviderConfig } from './config';

export interface MockStorageProvider extends FileStorageProvider {
  __list(): readonly UploadedFile[];
}

export function createMockStorageProvider(
  config: MockProviderConfig = DEFAULT_MOCK_CONFIG,
): MockStorageProvider {
  const files = new Map<string, UploadedFile>();

  return Object.freeze({
    name: 'storage' as const,
    mode: 'mock' as const,
    capabilities: ['uploadFile', 'deleteFile', 'getPublicUrl'] as const,
    async healthCheck(): Promise<HealthCheckResult> {
      return { status: 'ok', checkedAt: Date.now() };
    },
    async uploadFile(
      localUri: string,
      options?: FileUploadOptions,
    ): Promise<Result<UploadedFile>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      if (!localUri.trim()) {
        return { ok: false, error: AppError.validation({ code: 'EMPTY_URI' }) };
      }
      const id = `mock-file-${stableHash(localUri + Date.now())}`;
      const file: UploadedFile = Object.freeze({
        id,
        url: `memory://${id}`,
        size: localUri.length,
        contentType: options?.contentType ?? 'application/octet-stream',
        dataOrigin: 'demo' as const,
      });
      files.set(id, file);
      return { ok: true, value: file };
    },
    async deleteFile(fileId: string): Promise<Result<void>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      files.delete(fileId);
      return { ok: true, value: undefined };
    },
    async getPublicUrl(fileId: string): Promise<Result<string>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      const f = files.get(fileId);
      if (!f) return { ok: false, error: AppError.notFound({ code: 'FILE_NOT_FOUND' }) };
      return { ok: true, value: f.url };
    },
    __list(): readonly UploadedFile[] {
      return Array.from(files.values());
    },
  });
}
```

```
import { Text } from '../Text';
```

```
type RNTextInputRef = ElementRef<typeof RNTextInput>;

export interface OTPInputRef {
  focus(): void;
  clear(): void;
}

export interface OTPInputProps {
  /** Number of digits. Default 6. */
  readonly length?: number;
  /** Current digits (no separators). */
  readonly value: string;
  readonly onChangeText: (digitsOnly: string) => void;
  /** Called once when value.length === length. */
  readonly onComplete?: (code: string) => void;
  readonly label?: string;
  readonly required?: boolean;
  readonly helpText?: string;
  readonly errorText?: string;
  readonly error?: boolean;
  readonly disabled?: boolean;
  readonly autoFocus?: boolean;
  readonly testID?: string;
}

/**
 * OTP code input.
 *
 * Uses a single hidden TextInput overlaying N visual cells.
 * Benefits:
 *  - Paste works (multi-digit paste fills all cells).
 *  - iOS textContentType="oneTimeCode" auto-fill works.
 *  - Android autoComplete="sms-otp" works.
 *  - Backspace and cursor behave naturally.
 *  - One accessibility target — screen readers announce "verification code, N digits".
 */
export const OTPInput = forwardRef<OTPInputRef, OTPInputProps>(function OTPInput(
  {
    length = 6,
    value,
    onChangeText,
    onComplete,
    label,
    required = false,
    helpText,
    errorText,
    error = false,
    disabled = false,
    autoFocus = false,
    testID,
  },
  ref,
) {
  const theme = useTheme();
  const { isRTL } = useDirection();
  const [focused, setFocused] = useState(false);
  const inputRef = useRef<RNTextInputRef>(null);

  useImperativeHandle(
    ref,
    () => ({
      focus: () => inputRef.current?.focus(),
      clear: () => onChangeText(''),
    }),
    [onChangeText],
  );

  const hasError = error || !!errorText;
  const cells = Array.from({ length }, (_, i) => value[i] ?? '');
  const cursorIndex = Math.min(value.length, length - 1);

  const handleChange = (text: string) => {
    const digits = text.replace(/\D/g, '').slice(0, length);
    if (digits === value) return;
    onChangeText(digits);
    if (digits.length === length) {
      onComplete?.(digits);
    }
  };

  return (
    <Field
      label={label}
      required={required}
      helpText={helpText}
      errorText={errorText}
      disabled={disabled}
      testID={testID ? `${testID}-field` : undefined}
    >
      <View
        style={[
          styles.container,
          {
            gap: theme.spacing[2],
            flexDirection: isRTL ? 'row-reverse' : 'row',
            opacity: disabled ? 0.5 : 1,
          },
        ]}
      >
        {cells.map((char, i) => {
          const isCursor = focused && i === cursorIndex && value.length < length;
          const isFilled = char !== '';
          const borderColor = hasError
            ? theme.colors.state.danger
            : isCursor
              ? theme.colors.accent.fill
              : isFilled
                ? theme.colors.accent.fill
                : theme.colors.border.subtle;
          return (
            <View
              key={i}
              pointerEvents="none"
              style={[
                styles.cell,
                {
                  backgroundColor: theme.colors.bg.surface,
                  borderColor,
                  borderWidth: isCursor || hasError ? 2 : 1,
                  borderRadius: theme.radius.md,
                },
              ]}
            >
              {isCursor ? (
                <View style={[styles.cursor, { backgroundColor: theme.colors.accent.fill }]} />
              ) : (
                <Text variant="h2" color={isFilled ? 'primary' : 'tertiary'}>
                  {char || ''}
                </Text>
              )}
            </View>
          );
        })}

        <RNTextInput
          ref={inputRef}
          value={value}
          onChangeText={handleChange}
          onFocus={() => setFocused(true)}
          onBlur={() => setFocused(false)}
          editable={!disabled}
          autoFocus={autoFocus}
          keyboardType="number-pad"
          inputMode="numeric"
          maxLength={length}
          textContentType="oneTimeCode"
          autoComplete="sms-otp"
          caretHidden
          accessibilityLabel={label}
          accessibilityHint={`${length} ${length === 1 ? 'digit' : 'digits'}`}
          accessibilityState={{ disabled }}
          style={styles.hiddenInput}
          testID={testID}
        />
      </View>
    </Field>
  );
});

const styles = StyleSheet.create({
  container: {
    position: 'relative',
    alignItems: 'center',
  },
  cell: {
    flex: 1,
    aspectRatio: 1,
    maxWidth: 56,
    maxHeight: 56,
    alignItems: 'center',
    justifyContent: 'center',
  },

    position: 'absolute',
    top: 0,
    bottom: 0,
    left: 0,
    right: 0,
    opacity: 0.01,
    color: 'transparent',
  },
});
```

```
export function Checkbox({
  checked,

  indeterminate = false,
  disabled = false,
  label,
  error = false,
  testID,
```

```
// Avatar — initials-based display. Image URI support arrives in P06.
// Initials work for Arabic and Latin names using .charAt(0) — no
// localized strings rendered; caller provides the name.

import { StyleSheet, View, type StyleProp, type ViewStyle } from 'react-native';

import { useTheme } from '@/theme';

import { Text } from './Text';

export type AvatarSize = 'xs' | 'sm' | 'md' | 'lg' | 'xl';

const SIZES: Record<AvatarSize, { size: number; font: 'captionStrong' | 'bodyStrong' | 'h3' | 'h2' | 'h1' }> = {
  xs: { size: 24, font: 'captionStrong' },
  sm: { size: 32, font: 'captionStrong' },
  md: { size: 44, font: 'bodyStrong' },
  lg: { size: 64, font: 'h3' },
  xl: { size: 96, font: 'h1' },
};

export interface AvatarProps {
  readonly name: string;
  readonly size?: AvatarSize;
  /** Show a gold ring around the avatar. */
  readonly ring?: boolean;
  readonly style?: StyleProp<ViewStyle>;
  readonly testID?: string;
}

function getInitials(name: string): string {
  const words = name.trim().split(/\s+/).filter(Boolean);
  if (words.length === 0) return '?';
  if (words.length === 1) return words[0]!.charAt(0).toUpperCase();
  const first = words[0]!.charAt(0);
  const last = words[words.length - 1]!.charAt(0);
  return (first + last).toUpperCase();
}

export function Avatar({ name, size = 'md', ring = false, style, testID }: AvatarProps) {
  const theme = useTheme();
  const spec = SIZES[size];
  const initials = getInitials(name);
  const ringThickness = Math.max(2, Math.round(spec.size * 0.045));
  const outer = spec.size + ringThickness * 2 + 4;

  return (
    <View
      style={[
        styles.wrapper,
        {
          width: outer,
          height: outer,
          borderRadius: outer / 2,
          borderWidth: ring ? ringThickness : 0,
          borderColor: ring ? theme.colors.gold.fill : 'transparent',
          padding: ring ? 2 : 0,
        },
        style,
      ]}
      testID={testID}
      accessibilityRole="image"
      accessibilityLabel={name}
    >
      <View
        style={[
          styles.circle,
          {
            width: spec.size,
            height: spec.size,
            borderRadius: spec.size / 2,
            backgroundColor: theme.colors.accent.fill,
          },
        ]}
      >
        <Text variant={spec.font} color="primary">
          {initials}
        </Text>
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  wrapper: {
    alignItems: 'center',
    justifyContent: 'center',
  },
  circle: {
    alignItems: 'center',
    justifyContent: 'center',
  },
});
```

```
import type { ProfileRepository } from '@/repositories/contracts/profile';

export function profileRepositoryContract(getRepo: () => ProfileRepository): void {
  describe('ProfileRepository contract', () => {
    it('getById returns null for unknown id', async () => {
      const repo = getRepo();
      const r = await repo.getById('nonexistent');
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('getByUserId returns null when no profile exists', async () => {
      const repo = getRepo();
      const r = await repo.getByUserId('unknown-user');
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('create returns a profile with dataOrigin=demo', async () => {
      const repo = getRepo();
      const r = await repo.create({
        userId: 'user-1',
        displayName: 'Test Person',
        gender: 'male',
        maritalStatus: 'single',
      });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.userId).toBe('user-1');
        expect(r.value.dataOrigin).toBe('demo');
        expect(r.value.id).toBeTruthy();
        expect(r.value.createdAt).toBeTruthy();
        expect(r.value.updatedAt).toBeTruthy();
      }
    });

    it('rejects invalid input via Validation error', async () => {
      const repo = getRepo();
      const r = await repo.create({
        userId: '',
        displayName: '',
        gender: 'male',
        maritalStatus: 'single',
      });
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('Validation');
    });

    it('created profile can be retrieved by id and userId', async () => {
      const repo = getRepo();
      const created = await repo.create({
        userId: 'user-2',
        displayName: 'Retrieve Me',
        gender: 'female',
        maritalStatus: 'single',
        city: 'Sanaa',
      });
      if (!created.ok) throw new Error('setup failed');
      const byId = await repo.getById(created.value.id);
      expect(byId.ok).toBe(true);
      if (byId.ok && byId.value) expect(byId.value.id).toBe(created.value.id);
      const byUser = await repo.getByUserId('user-2');
      expect(byUser.ok).toBe(true);
      if (byUser.ok && byUser.value) expect(byUser.value.userId).toBe('user-2');
    });

    it('update modifies fields and updatedAt', async () => {
      const repo = getRepo();
      const created = await repo.create({
        userId: 'u3',
        displayName: 'Before',
        gender: 'male',
        maritalStatus: 'single',
      });
      if (!created.ok) throw new Error('setup failed');
      const r = await repo.update(created.value.id, { displayName: 'After' });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.displayName).toBe('After');
        expect(r.value.updatedAt >= created.value.updatedAt).toBe(true);
      }
    });

    it('update returns NotFound for unknown id', async () => {
      const repo = getRepo();
      const r = await repo.update('nope', { displayName: 'X' });
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('NotFound');
    });

    it('delete removes the profile', async () => {
      const repo = getRepo();
      const created = await repo.create({
        userId: 'u4',
        displayName: 'Delete Me',
        gender: 'male',
        maritalStatus: 'single',
      });
      if (!created.ok) throw new Error('setup failed');
      const del = await repo.delete(created.value.id);
      expect(del.ok).toBe(true);
      const r = await repo.getById(created.value.id);
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('list returns PagedResult and respects filters', async () => {
      const repo = getRepo();
      await repo.create({
        userId: 'a',
        displayName: 'A',
        gender: 'male',
        maritalStatus: 'single',
        city: 'Sanaa',
      });
      await repo.create({
        userId: 'b',
        displayName: 'B',
        gender: 'male',
        maritalStatus: 'married',
        city: 'Aden',
      });
      const r = await repo.list({ city: 'Sanaa' });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.total).toBe(1);
        expect(r.value.items[0]?.city).toBe('Sanaa');
      }
    });

    it('list respects limit/offset', async () => {
      const repo = getRepo();
      for (let i = 0; i < 5; i += 1) {
        await repo.create({
          userId: `u${i}`,
          displayName: `P${i}`,
          gender: 'unspecified',
          maritalStatus: 'single',
        });
      }
      const page = await repo.list({ limit: 2, offset: 1 });
      expect(page.ok).toBe(true);
      if (page.ok) {
        expect(page.value.items.length).toBe(2);
        expect(page.value.total).toBe(5);
        expect(page.value.limit).toBe(2);
        expect(page.value.offset).toBe(1);
      }
    });
  });
}
```
