import type { Result } from '@/core/result';

import type { BaseEntity, ListQuery, PagedResult } from '../types';

export type ApplicationStatus =
  | 'draft'
  | 'submitted'
  | 'reviewing'
  | 'approved'
  | 'rejected'
  | 'disbursed';

export interface Application extends BaseEntity {
  readonly applicantId: string;
  readonly programId: string;
  readonly status: ApplicationStatus;
  readonly requestedAmount: number;
  readonly currency: 'YER';
  readonly notes: string | null;
  readonly reviewerNote: string | null;
}

export interface CreateApplicationInput {
  readonly applicantId: string;
  readonly programId: string;
  readonly requestedAmount: number;
  readonly notes?: string | null;
}

export interface ApplicationListQuery extends ListQuery {
  readonly status?: ApplicationStatus;
  readonly programId?: string;
  readonly applicantId?: string;
}

export interface ApplicationRepository {
  getById(id: string): Promise<Result<Application | null>>;
  list(query?: ApplicationListQuery): Promise<Result<PagedResult<Application>>>;
  listByApplicant(applicantId: string): Promise<Result<readonly Application[]>>;
  create(input: CreateApplicationInput): Promise<Result<Application>>;
  updateStatus(
    id: string,
    status: ApplicationStatus,
    reviewerNote?: string,
  ): Promise<Result<Application>>;
  delete(id: string): Promise<Result<void>>;
}
```

import { AppError } from '@/core/errors';
import { ok, type Result } from '@/core/result';
import { CreateApplicationInputSchema } from '@/schemas/application';

import type {
  Application,
  ApplicationListQuery,
  ApplicationRepository,
  ApplicationStatus,
  CreateApplicationInput,
} from '../contracts/application';
import type { PagedResult } from '../types';
import type { MockRepositoryOptions } from './profile';

export function createMockApplicationRepository(
  options: MockRepositoryOptions = {},
): ApplicationRepository {
  const { delayMs = 0, forceErrorKind = null } = options;
  const byId = new Map<string, Application>();

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    return null;
  }

  return Object.freeze({
    async getById(id: string): Promise<Result<Application | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      return ok(byId.get(id) ?? null);
    },

    async list(query?: ApplicationListQuery): Promise<Result<PagedResult<Application>>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const limit = query?.limit ?? 20;
      const offset = query?.offset ?? 0;
      let items: Application[] = Array.from(byId.values());
      if (query?.status) items = items.filter((a) => a.status === query.status);
      if (query?.programId) items = items.filter((a) => a.programId === query.programId);
      if (query?.applicantId) items = items.filter((a) => a.applicantId === query.applicantId);
      const total = items.length;
      items = items.slice(offset, offset + limit);
      return ok({ items, total, limit, offset });
    },

    async listByApplicant(applicantId: string): Promise<Result<readonly Application[]>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const items = Array.from(byId.values()).filter((a) => a.applicantId === applicantId);
      return ok(items);
    },

    async create(input: CreateApplicationInput): Promise<Result<Application>> {
      const parsed = CreateApplicationInputSchema.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const id = `mock-application-${Date.now()}-${byId.size}`;
      const now = new Date().toISOString();
      const app: Application = Object.freeze({
        id,
        applicantId: parsed.data.applicantId,
        programId: parsed.data.programId,
        status: 'draft' as const,
        requestedAmount: parsed.data.requestedAmount,
        currency: 'YER' as const,
        notes: parsed.data.notes ?? null,
        reviewerNote: null,
        createdAt: now,
        updatedAt: now,
        dataOrigin: 'demo' as const,
      });
      byId.set(id, app);
      return ok(app);
    },

    async updateStatus(
      id: string,
      status: ApplicationStatus,
      reviewerNote?: string,
    ): Promise<Result<Application>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const existing = byId.get(id);
      if (!existing) {
        return { ok: false, error: AppError.notFound({ code: 'APPLICATION_NOT_FOUND' }) };
      }
      const updated: Application = Object.freeze({
        ...existing,
        status,
        ...(reviewerNote !== undefined && { reviewerNote }),
        updatedAt: new Date().toISOString(),
      });
      byId.set(id, updated);
      return ok(updated);
    },

    async delete(id: string): Promise<Result<void>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      byId.delete(id);
      return ok(undefined);
    },
  });
}

// Single entry point for auth actions. Screens never call providers directly.
import { useCallback, useEffect, useMemo, useState } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import type { AppError } from '@/core/errors';
import type { Result } from '@/core/result';
import type {
  AuthChallenge,
  AuthSession,
  AuthUser,
  ResetPasswordInput,
  RoleIntent,
  SignInCredentials,
} from '@/providers';
import { useProviderContainer } from '@/providers';
import { useSession } from '@/features/shell/use-session';
import { useAppModeStore, useAuthStore } from '@/stores';
import { deriveAuthPhase, type AuthPhase } from './auth-machine';
import { buildConsents } from './consents';
import { createTokenVault } from './token-vault';

export interface SignUpPayload extends SignInCredentials {
  readonly displayName: string;
  readonly roleIntent: RoleIntent;
}

export interface AuthApi {
  readonly phase: AuthPhase;
  readonly user: AuthUser | null;
  readonly session: AuthSession | null;
  readonly isGuest: boolean;
  readonly busy: boolean;
  readonly error: AppError | null;
  readonly channels: readonly ('email' | 'phone')[];
  signIn(credentials: SignInCredentials): Promise<Result<AuthSession>>;
  signUp(payload: SignUpPayload): Promise<Result<AuthSession>>;
  verify(code: string): Promise<Result<AuthSession>>;
  resend(): Promise<Result<AuthChallenge>>;
  requestReset(email: string): Promise<Result<AuthChallenge>>;
  resetPassword(code: string, password: string): Promise<Result<void>>;
  continueAsGuest(): void;
  signOut(): Promise<Result<void>>;
  clearError(): void;
  refetch(): void;
}

const REFRESH_MARGIN_MS = 60_000;

export function useAuth(): AuthApi {
  const container = useProviderContainer();
  const session = useSession();
  const client = useQueryClient();
  const modeHydrated = useAppModeStore((s) => s.hydrated);
  const authHydrated = useAuthStore((s) => s.hydrated);
  const guestActive = useAuthStore((s) => s.guestActive);
  const challenge = useAuthStore((s) => s.challenge);
  const startGuest = useAuthStore((s) => s.startGuest);
  const endGuest = useAuthStore((s) => s.endGuest);
  const setChallenge = useAuthStore((s) => s.setChallenge);

  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<AppError | null>(null);
  const vault = useMemo(() => createTokenVault(), []);

  const resolved = session.result?.ok ? session.result.value : null;
  const setSessionData = session.setSession;

  const phase = deriveAuthPhase({
    hydrated: modeHydrated && authHydrated,
    guestActive,
    sessionLoading: session.isLoading,
    session: session.result,
  });

  const run = useCallback(async <T,>(fn: () => Promise<Result<T>>): Promise<Result<T>> => {
    setBusy(true);
    setError(null);
    const result = await fn();
    if (!result.ok) setError(result.error);
    setBusy(false);
    return result;
  }, []);

  const persist = useCallback(
    async (value: AuthSession): Promise<void> => {
      if (value.tokens) await vault.save(value.tokens.refreshToken);
    },
    [vault],
  );

  const storeChallenge = useCallback(
    (value: AuthChallenge): void => {
      setChallenge({
        id: value.id,
        channel: value.channel,
        target: value.target,
        expiresAt: value.expiresAt,
        demoCode: value.demoCode ?? null,
      });
    },
    [setChallenge],
  );

  const signIn = useCallback(
    async (credentials: SignInCredentials) => {
      const result = await run(() => container.auth.signIn(credentials));
      if (result.ok) {
        endGuest();
        await persist(result.value);
        setSessionData(result.value);
      }
      return result;
    },
    [container, run, endGuest, persist, setSessionData],
  );

  const signUp = useCallback(
    async (payload: SignUpPayload) => {
      const result = await run(() =>
        container.auth.signUp({
          displayName: payload.displayName,
          email: payload.email,
          password: payload.password,
          roleIntent: payload.roleIntent,
          consents: buildConsents(),
        }),
      );
      if (result.ok) {
        endGuest();
        await persist(result.value);
        setSessionData(result.value);
        const pending = result.value.pendingVerification;
        if (pending?.challengeId) {
          setChallenge({
            id: pending.challengeId,
            channel: pending.channel,
            target: pending.target,
            expiresAt: new Date(Date.now() + 10 * 60 * 1000).toISOString(),
            demoCode: pending.demoCode ?? null,
          });
        }
      }
      return result;
    },
    [container, run, endGuest, persist, setSessionData, setChallenge],
  );

  const verify = useCallback(
    async (code: string) => {
      const result = await run(() =>
        container.auth.verify({ code, challengeId: challenge?.id }),
      );
      if (result.ok) {
        await persist(result.value);
        setSessionData(result.value);
        setChallenge(null);
      }
      return result;
    },
    [container, run, challenge, persist, setSessionData, setChallenge],
  );

  const resend = useCallback(async () => {
    const target = resolved?.pendingVerification?.target ?? challenge?.target;
    if (!target) {
      const issued = await run(() =>
        container.auth.requestVerification({ channel: 'email', target: '' }),
      );
      return issued;
    }
    const result = await run(() =>
      container.auth.requestVerification({ channel: 'email', target }),
    );
    if (result.ok) storeChallenge(result.value);
    return result;
  }, [container, run, resolved, challenge, storeChallenge]);

  const requestReset = useCallback(
    async (email: string) => {
      const result = await run(() => container.auth.requestPasswordReset({ email }));
      if (result.ok) storeChallenge(result.value);
      return result;
    },
    [container, run, storeChallenge],
  );

  const resetPassword = useCallback(
    async (code: string, password: string) => {
      const input: ResetPasswordInput = {
        challengeId: challenge?.id ?? '',
        code,
        password,
      };
      const result = await run(() => container.auth.resetPassword(input));
      if (result.ok) setChallenge(null);
      return result;
    },
    [container, run, challenge, setChallenge],
  );

  const continueAsGuest = useCallback((): void => {
    setError(null);
    startGuest();
    setSessionData(null);
  }, [startGuest, setSessionData]);

  const signOut = useCallback(async () => {
    const result = await run(() => container.auth.signOut());
    if (result.ok) {
      await vault.clear();
      endGuest();
      setChallenge(null);
      setSessionData(null);
      client.clear();
    }
    return result;
  }, [run, container, vault, endGuest, setChallenge, setSessionData, client]);

  const clearError = useCallback(() => setError(null), []);

  // Automatic refresh, scheduled only when the session actually carries tokens.
  useEffect(() => {
    if (!resolved?.tokens) return;
    const delay = Math.max(
      5_000,
      new Date(resolved.expiresAt).getTime() - Date.now() - REFRESH_MARGIN_MS,
    );
    const timer = setTimeout(() => {
      void container.auth.refreshSession().then((result) => {
        if (result.ok && result.value) setSessionData(result.value);
      });
    }, delay);
    return () => clearTimeout(timer);
  }, [resolved, container, setSessionData]);

  return {
    phase,
    user: resolved?.user ?? null,
    session: resolved,
    isGuest: phase === 'guest',
    busy,
    error,
    channels: container.auth.channels,
    signIn,
    signUp,
    verify,
    resend,
    requestReset,
    resetPassword,
    continueAsGuest,
    signOut,
    clearError,
    refetch: session.refetch,
  };
}

// Locale-aware formatters. Uses Intl directly (available on Node 26, browsers, Hermes RN 0.88).
// Fallbacks only trigger if Intl or a locale is unavailable; warnings are logged, not thrown.

import { logger } from '@/core/logger';

import type { SupportedLanguage } from './config';

export type NumeralSystem = 'latn' | 'arab';
export type DateStyle = 'short' | 'medium' | 'long';

export interface FormatOptions {
  readonly language?: SupportedLanguage;
  readonly numerals?: NumeralSystem;
}

const DEFAULT_LANGUAGE: SupportedLanguage = 'ar';
const DEFAULT_NUMERALS_AR: NumeralSystem = 'arab';
const DEFAULT_NUMERALS_EN: NumeralSystem = 'latn';
const CURRENCY_CODE = 'YER';

function resolve(language: SupportedLanguage | undefined, numerals: NumeralSystem | undefined): {
  language: SupportedLanguage;
  numerals: NumeralSystem;
} {
  const lang = language ?? DEFAULT_LANGUAGE;
  const nums = numerals ?? (lang === 'ar' ? DEFAULT_NUMERALS_AR : DEFAULT_NUMERALS_EN);
  return { language: lang, numerals: nums };
}

export function isIntlAvailable(): boolean {
  return (
    typeof Intl !== 'undefined' &&
    typeof Intl.NumberFormat === 'function' &&
    typeof Intl.DateTimeFormat === 'function'
  );
}

export function formatNumber(value: number, opts: FormatOptions = {}): string {
  const { language, numerals } = resolve(opts.language, opts.numerals);
  if (!isIntlAvailable()) return String(value);
  try {
    return new Intl.NumberFormat(language, { numberingSystem: numerals }).format(value);
  } catch (e) {
    logger.warn('formatNumber failed', { error: e });
    return String(value);
  }
}

export function formatCurrency(amount: number, opts: FormatOptions = {}): string {
  const { language, numerals } = resolve(opts.language, opts.numerals);
  if (!isIntlAvailable()) {
    const suffix = language === 'ar' ? 'ر.ي' : 'YER';// i18n-allow-arabic
    return `${String(amount)} ${suffix}`;
  }
  try {
    return new Intl.NumberFormat(language, {
      style: 'currency',
      currency: CURRENCY_CODE,
      maximumFractionDigits: 0,
      minimumFractionDigits: 0,
      numberingSystem: numerals,
    }).format(amount);
  } catch (e) {
    logger.warn('formatCurrency failed', { error: e });
    const suffix = language === 'ar' ? 'ر.ي' : 'YER';// i18n-allow-arabic
    return `${String(amount)} ${suffix}`;
  }
}

export function formatDate(
  input: Date | string | number,
  style: DateStyle = 'medium',
  opts: FormatOptions = {},
): string {
  const date = input instanceof Date ? input : new Date(input);
  if (Number.isNaN(date.getTime())) {
    logger.warn('formatDate received invalid date', { input: String(input) });
    return '';
  }
  const { language, numerals } = resolve(opts.language, opts.numerals);
  if (!isIntlAvailable()) {
    return date.toISOString().slice(0, 10);
  }
  try {
    return new Intl.DateTimeFormat(language, {
      dateStyle: style,
      numberingSystem: numerals,
    }).format(date);
  } catch (e) {
    logger.warn('formatDate failed', { error: e });
    return date.toISOString().slice(0, 10);
  }
}

export function formatPercent(ratio: number, opts: FormatOptions = {}): string {
  const { language, numerals } = resolve(opts.language, opts.numerals);
  if (!isIntlAvailable()) return `${String(Math.round(ratio * 100))}%`;
  try {
    return new Intl.NumberFormat(language, {
      style: 'percent',
      maximumFractionDigits: 1,
      numberingSystem: numerals,
    }).format(ratio);
  } catch (e) {
    logger.warn('formatPercent failed', { error: e });
    return `${String(Math.round(ratio * 100))}%`;
  }
}

  readonly expiresAt: string;

  readonly challengeId?: string;

  readonly demoCode?: string;

  readonly id: string;
  readonly email: string | null;
  readonly phone: string | null;
  readonly displayName: string;

  readonly dataOrigin: 'real' | 'demo';
  /** ISO 8601. */
  readonly createdAt: string;
  /** Absent until P06 clients populate it; deny-by-default in guards. */
  readonly roles?: readonly AuthRole[];
  readonly consents?: readonly AuthConsent[];

  readonly expiresAt: string;

  readonly policyStatus: 'pending-human-approval';

/** Everything the account owner may export. Never contains tokens or passwords. */
export interface AccountExport {
  /** ISO 8601. */
  readonly generatedAt: string;
  readonly dataOrigin: 'real' | 'demo';

  readonly consents: readonly AuthConsent[];

  requestAccountDeletion(input: DeleteAccountInput): Promise<Result<AccountDeletionRequest>>;

  exportAccountData(): Promise<Result<AccountExport>>;
