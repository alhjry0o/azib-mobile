import { Cairo_600SemiBold, Cairo_700Bold } from '@expo-google-fonts/cairo';
import {
  IBMPlexSansArabic_400Regular,
  IBMPlexSansArabic_500Medium,
  IBMPlexSansArabic_600SemiBold,
  IBMPlexSansArabic_700Bold,
} from '@expo-google-fonts/ibm-plex-sans-arabic';
import { useFonts } from 'expo-font';
import { Stack } from 'expo-router';
import { StatusBar } from 'expo-status-bar';
import { I18nManager, StyleSheet, View } from 'react-native';
import { I18nextProvider } from 'react-i18next';
import { SafeAreaProvider } from 'react-native-safe-area-context';

import { ToastProvider } from '@/components/ui';
import { i18n } from '@/i18n';
import { ProviderContainerProvider } from '@/providers';
import { ThemeProvider } from '@/theme';

I18nManager.allowRTL(true);

export default function RootLayout() {
  const [fontsLoaded, fontError] = useFonts({
    IBMPlexSansArabic_400Regular,
    IBMPlexSansArabic_500Medium,
    IBMPlexSansArabic_600SemiBold,
    IBMPlexSansArabic_700Bold,
    Cairo_600SemiBold,
    Cairo_700Bold,
  });

  if (!fontsLoaded && !fontError) {
    return <View style={styles.root} />;
  }

  return (
    <I18nextProvider i18n={i18n}>
      <ThemeProvider>
        <SafeAreaProvider>
          <ToastProvider>
            <ProviderContainerProvider>
              <View style={styles.root}>
                <StatusBar style="light" />
                <Stack
                  screenOptions={{
                    headerShown: false,
                    contentStyle: styles.content,
                  }}
                />
              </View>
            </ProviderContainerProvider>
          </ToastProvider>
        </SafeAreaProvider>
      </ThemeProvider>
    </I18nextProvider>
  );
}

const styles = StyleSheet.create({
  root: {
    flex: 1,
    backgroundColor: '#09080F',
  },
  content: {
    backgroundColor: '#09080F',
  },
});

/**
 * env — الإعدادات العامة (EXPO_PUBLIC_*) مع تحقق Zod وفشل مبكر.
 *
 * ما هو هنا: متغيرات آمنة للنشر (تبدأ بـ EXPO_PUBLIC_*).
 * ما ليس هنا: أي سر (service_role، مفاتيح دفع، إلخ) — تلك لا تصل العميل أصلاً.
 *
 * إن غاب متغير مطلوب أو كان غير صحيح، نرمي فوراً برسالة واضحة
 * لأن التطبيق لا يمكن أن يعمل في حالة غامضة.
 */

import { z } from 'zod';

const RawEnvSchema = z.object({
  EXPO_PUBLIC_APP_MODE: z.enum(['development', 'staging', 'production']).default('development'),
  EXPO_PUBLIC_ENV: z.enum(['local', 'dev', 'staging', 'prod']).default('local'),
  EXPO_PUBLIC_SUPABASE_URL: z.string().trim().optional().or(z.literal('')),
  EXPO_PUBLIC_SUPABASE_ANON_KEY: z.string().trim().optional().or(z.literal('')),
});

export type RawEnvInput = z.infer<typeof RawEnvSchema>;

export interface AppEnv {
  readonly APP_MODE: 'development' | 'staging' | 'production';
  readonly ENV: 'local' | 'dev' | 'staging' | 'prod';
  readonly SUPABASE_URL: string | null;
  readonly SUPABASE_ANON_KEY: string | null;
  /** true فقط عندما يكون كلا المتغيرين موجودين معاً وصحيحَين. */
  readonly isSupabaseConfigured: boolean;
}

export class EnvValidationError extends Error {
  readonly issues: readonly string[];
  constructor(issues: readonly string[]) {
    super(`Invalid environment variables:\n - ${issues.join('\n - ')}`);
    this.name = 'EnvValidationError';
    this.issues = issues;
    Object.freeze(this.issues);
  }
}

/**
 * parseEnv — دالة نقية قابلة للاختبار.
 * تمرير `process.env` أو كائن اختباري.
 */
export function parseEnv(source: Record<string, string | undefined>): AppEnv {
  const result = RawEnvSchema.safeParse(source);
  if (!result.success) {
    const issues = result.error.issues.map((i) => `${i.path.join('.')}: ${i.message}`);
    throw new EnvValidationError(issues);
  }

  const { EXPO_PUBLIC_APP_MODE, EXPO_PUBLIC_ENV, EXPO_PUBLIC_SUPABASE_URL, EXPO_PUBLIC_SUPABASE_ANON_KEY } =
    result.data;

  const url = EXPO_PUBLIC_SUPABASE_URL && EXPO_PUBLIC_SUPABASE_URL.length > 0 ? EXPO_PUBLIC_SUPABASE_URL : null;
  const key = EXPO_PUBLIC_SUPABASE_ANON_KEY && EXPO_PUBLIC_SUPABASE_ANON_KEY.length > 0 ? EXPO_PUBLIC_SUPABASE_ANON_KEY : null;

  // إذا وُجد أحدهما فقط، الإعداد ناقص — فشل مبكر بدل تشغيل بنصف إعداد.
  if ((url && !key) || (!url && key)) {
    throw new EnvValidationError([
      'EXPO_PUBLIC_SUPABASE_URL and EXPO_PUBLIC_SUPABASE_ANON_KEY must be set together or both absent.',
    ]);
  }

  return Object.freeze({
    APP_MODE: EXPO_PUBLIC_APP_MODE,
    ENV: EXPO_PUBLIC_ENV,
    SUPABASE_URL: url,
    SUPABASE_ANON_KEY: key,
    isSupabaseConfigured: url !== null && key !== null,
  });
}

/**
 * env — القيمة المحسوبة مرة واحدة عند الاستيراد.
 * أي متغير غير صحيح يُفشل التطبيق عند البدء برسالة واضحة.
 */
export const env: AppEnv = parseEnv(process.env as Record<string, string | undefined>);
```

// ProviderContainerProvider — React context that holds a single container
// and rebuilds it when the AppMode changes.
//
// Usage in app/_layout.tsx:
//   <ProviderContainerProvider><Root /></ProviderContainerProvider>
//
// Usage in a component:
//   const { auth, payment, registry } = useProviderContainer();

import { createContext, useContext, useMemo, type ReactNode, type ReactElement } from 'react';

import { env } from '@/config/env';
import { useAppModeStore } from '@/stores/app-mode.store';

import { createProviderContainer, type ProviderContainer } from './container';

const ProviderContainerContext = createContext<ProviderContainer | null>(null);

export interface ProviderContainerProviderProps {
  readonly children: ReactNode;
}

export function ProviderContainerProvider({
  children,
}: ProviderContainerProviderProps): ReactElement {
  const mode = useAppModeStore((s) => s.mode);

  const container = useMemo<ProviderContainer>(
    () => createProviderContainer({ mode, env }),
    [mode],
  );

  return (
    <ProviderContainerContext.Provider value={container}>
      {children}
    </ProviderContainerContext.Provider>
  );
}

export function useProviderContainer(): ProviderContainer {
  const c = useContext(ProviderContainerContext);
  if (!c) {
    throw new Error(
      'useProviderContainer must be used within <ProviderContainerProvider>. ' +
        'Wrap your root layout with it.',
    );
  }
  return c;
}

```
// IntegrationRegistry tracks the runtime status of every provider.
// The UI (Developer Settings, P05) reads this to display connection state.
// The ProviderContainer (Part 3b) populates it on init.

import { ALL_PROVIDER_NAMES, type HealthCheckResult, type IntegrationRecord, type ProviderName, type ProviderStatus } from './types';

export class IntegrationRegistry {
  private readonly records: Map<ProviderName, IntegrationRecord>;

  constructor() {
    this.records = new Map();
    this.reset();
  }

  reset(): void {
    this.records.clear();
    for (const name of ALL_PROVIDER_NAMES) {
      this.records.set(name, Object.freeze({ name, status: 'disabled' as ProviderStatus }));
    }
  }

  setStatus(name: ProviderName, status: ProviderStatus): void {
    const prev = this.records.get(name);
    this.records.set(
      name,
      Object.freeze({
        name,
        status,
        healthCheck: prev?.healthCheck,
        lastCheckedAt: prev?.lastCheckedAt,
      }),
    );
  }

  setHealthCheck(name: ProviderName, result: HealthCheckResult): void {
    const prev = this.records.get(name);
    if (!prev) {
      throw new Error(`IntegrationRegistry: unknown provider "${name}"`);
    }
    let status: ProviderStatus = prev.status;
    if (result.status === 'ok') status = 'connected';
    else if (result.status === 'degraded') status = 'connected';
    else if (result.status === 'unavailable') {
      if (prev.status === 'mock') status = 'mock';
      else status = 'error';
    }
    this.records.set(
      name,
      Object.freeze({
        name,
        status,
        healthCheck: result,
        lastCheckedAt: result.checkedAt,
      }),
    );
  }

  get(name: ProviderName): IntegrationRecord {
    const r = this.records.get(name);
    if (!r) throw new Error(`IntegrationRegistry: unknown provider "${name}"`);
    return r;
  }

  list(): readonly IntegrationRecord[] {
    return ALL_PROVIDER_NAMES.map((n) => this.get(n));
  }

  snapshot(): Readonly<Record<ProviderName, IntegrationRecord>> {
    const out = {} as Record<ProviderName, IntegrationRecord>;
    for (const name of ALL_PROVIDER_NAMES) {
      out[name] = this.get(name);
    }
    return Object.freeze(out);
  }
}

/** Global registry. Populated by the ProviderContainer. */
export const integrationRegistry = new IntegrationRegistry();

/** Factory for tests to get an isolated registry. */
export function createIntegrationRegistry(): IntegrationRegistry {
  return new IntegrationRegistry();
}

// Sign-in accepts any non-empty email + password (demo mode is honest: no real auth).

import { AppError } from '@/core/errors';
import type { Result } from '@/core/result';

  AuthProvider,
  AuthSession,
  AuthUser,
  SignInCredentials,
} from '../contracts/auth';
import type { HealthCheckResult } from '../types';
import { DEFAULT_MOCK_CONFIG, simulateMockIO, stableHash, type MockProviderConfig } from './config';

const SESSION_TTL_MS = 60 * 60 * 1000;

function makeUser(email: string): AuthUser {
  const id = `demo-user-${stableHash(email)}`;
  return Object.freeze({
    id,
    email,
    phone: null,
    displayName: 'Demo User',
    dataOrigin: 'demo' as const,
    createdAt: new Date().toISOString(),
  });
}

function makeSession(user: AuthUser): AuthSession {
  return Object.freeze({
    user,
    expiresAt: new Date(Date.now() + SESSION_TTL_MS).toISOString(),
  });
}

export function createMockAuthProvider(
  config: MockProviderConfig = DEFAULT_MOCK_CONFIG,
): AuthProvider {
  let session: AuthSession | null = null;

  return Object.freeze({
    name: 'auth' as const,
    mode: 'mock' as const,
    capabilities: ['signIn', 'signOut', 'getSession', 'refreshSession'] as const,
    async healthCheck(): Promise<HealthCheckResult> {
      return { status: 'ok', checkedAt: Date.now() };
    },
    async signIn(credentials: SignInCredentials): Promise<Result<AuthSession>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      if (!credentials.email.trim() || !credentials.password.trim()) {
        return { ok: false, error: AppError.validation({ code: 'EMPTY_CREDENTIALS' }) };
      }
      session = makeSession(makeUser(credentials.email.trim()));
      return { ok: true, value: session };
    },
    async signOut(): Promise<Result<void>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      session = null;
      return { ok: true, value: undefined };
    },
    async getSession(): Promise<Result<AuthSession | null>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      return { ok: true, value: session };
    },
    async refreshSession(): Promise<Result<AuthSession | null>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      if (!session) return { ok: true, value: null };
      session = makeSession(session.user);
      return { ok: true, value: session };
    },
  });
}

/**
 * AppError — تصنيف موحّد لأخطاء التطبيق.
 *
 * القواعد:
 * - `kind` هو التصنيف العام (للاتخاذ الآلي).
 * - `i18nKey` هو مفتاح الترجمة للعرض للمستخدم.
 * - `meta` للمعلومات التشخيصية (لا تُعرض للمستخدم مباشرة).
 * - لا ترمِ AppError مباشرة من داخل components. أعِدها كـ Result.
 */

export type AppErrorKind =
  | 'Network'
  | 'Auth'
  | 'Forbidden'
  | 'Validation'
  | 'NotFound'
  | 'Conflict'
  | 'NotConnected'
  | 'Unknown';

export interface AppErrorMeta {
  /** كود داخلي اختياري (مثل 'PROFILE_MISSING'). */
  readonly code?: string;
  /** حالة HTTP إن أتت من طبقة شبكة. */
  readonly httpStatus?: number;
  /** حقول إضافية تشخيصية — لا تُعرض للمستخدم. */
  readonly context?: Readonly<Record<string, unknown>>;
  /** السبب الأصلي (خطأ JS أو AppError آخر). */
  readonly cause?: unknown;
}

const I18N_KEYS: Record<AppErrorKind, string> = {
  Network: 'errors.network',
  Auth: 'errors.auth',
  Forbidden: 'errors.forbidden',
  Validation: 'errors.validation',
  NotFound: 'errors.notFound',
  Conflict: 'errors.conflict',
  NotConnected: 'errors.notConnected',
  Unknown: 'errors.unknown',
};

export class AppError {
  readonly kind: AppErrorKind;
  readonly i18nKey: string;
  readonly meta: AppErrorMeta;

  private constructor(kind: AppErrorKind, meta: AppErrorMeta) {
    this.kind = kind;
    this.i18nKey = I18N_KEYS[kind];
    this.meta = meta;
    Object.freeze(this.meta);
    Object.freeze(this);
  }

  static network(meta: AppErrorMeta = {}): AppError {
    return new AppError('Network', meta);
  }
  static auth(meta: AppErrorMeta = {}): AppError {
    return new AppError('Auth', meta);
  }
  static forbidden(meta: AppErrorMeta = {}): AppError {
    return new AppError('Forbidden', meta);
  }
  static validation(meta: AppErrorMeta = {}): AppError {
    return new AppError('Validation', meta);
  }
  static notFound(meta: AppErrorMeta = {}): AppError {
    return new AppError('NotFound', meta);
  }
  static conflict(meta: AppErrorMeta = {}): AppError {
    return new AppError('Conflict', meta);
  }
  static notConnected(meta: AppErrorMeta = {}): AppError {
    return new AppError('NotConnected', meta);
  }
  static unknown(meta: AppErrorMeta = {}): AppError {
    return new AppError('Unknown', meta);
  }

  static fromKind(kind: AppErrorKind, meta: AppErrorMeta = {}): AppError {
    return new AppError(kind, meta);
  }

  /** للعرض في سجلات التشخيص فقط. لا تُعرض للمستخدم. */
  toString(): string {
    return `AppError(${this.kind}, code=${this.meta.code ?? '-'})`;
  }

  toJSON(): {
    kind: AppErrorKind;
    i18nKey: string;
    code: string | null;
    httpStatus: number | null;
    context: Record<string, unknown> | null;
  } {
    return {
      kind: this.kind,
      i18nKey: this.i18nKey,
      code: this.meta.code ?? null,
      httpStatus: this.meta.httpStatus ?? null,
      context: this.meta.context ? { ...this.meta.context } : null,
    };
  }
}

/** يحوّل أي `unknown` مُلتقَط إلى AppError. مفيد في catch blocks. */
export function toAppError(caught: unknown): AppError {
  if (caught instanceof AppError) return caught;

  if (caught instanceof Error) {
    const msg = caught.message.toLowerCase();
    if (msg.includes('network') || msg.includes('fetch')) {
      return AppError.network({ cause: caught });
    }
    return AppError.unknown({ cause: caught });
  }

  return AppError.unknown({ cause: caught });
}

/**
 * Result<T, E> — نمط بديل للاستثناءات في حدود الإجراءات الداخلية.
 *
 * القاعدة: استخدم Result داخل repositories و adapters.
 * لا تستخدم Result داخل React components مباشرة (استخدم isLoading/error).
 *
 * `E` افتراضياً هو `AppError`، لكنه عام لاستخدامه مع أنواع أخرى في المستقبل.
 */

import type { AppError } from './errors';

export type Result<T, E = AppError> =
  | { readonly ok: true; readonly value: T }
  | { readonly ok: false; readonly error: E };

export function ok<T>(value: T): Result<T, never> {
  return { ok: true, value };
}

export function err<E>(error: E): Result<never, E> {
  return { ok: false, error };
}

export function isOk<T, E>(r: Result<T, E>): r is { ok: true; value: T } {
  return r.ok;
}

export function isErr<T, E>(r: Result<T, E>): r is { ok: false; error: E } {
  return !r.ok;
}

export function map<T, U, E>(r: Result<T, E>, fn: (value: T) => U): Result<U, E> {
  return r.ok ? { ok: true, value: fn(r.value) } : r;
}

export function mapErr<T, E, F>(r: Result<T, E>, fn: (error: E) => F): Result<T, F> {
  return r.ok ? r : { ok: false, error: fn(r.error) };
}

export function unwrapOr<T, E>(r: Result<T, E>, fallback: T): T {
  return r.ok ? r.value : fallback;
}

/**
 * mapAsync — يأخذ قيمة T ويعيد Promise<U>، ويغلّفها في Result.
 * الاستخدام: سلسلة async operations داخل service دون رمي استثناءات.
 */
export async function mapAsync<T, U, E>(
  r: Result<T, E>,
  fn: (value: T) => Promise<U>,
): Promise<Result<U, E>> {
  if (!r.ok) return r;
  return { ok: true, value: await fn(r.value) };
}

/**
 * tryAsync — يشغّل دالة async ويعيد Result.
 * الأخطاء التي تُرمى تُحوَّل عبر `onError` (يجب أن ينتج AppError أو نوع E).
 */
export async function tryAsync<T, E>(
  fn: () => Promise<T>,
  onError: (caught: unknown) => E,
): Promise<Result<T, E>> {
  try {
    return { ok: true, value: await fn() };
  } catch (caught) {
    return { ok: false, error: onError(caught) };
  }
}

// Device tier determines which visual effects run.
// Low tier: no glows, no shadows, no parallax. Mid: shadows only. High: everything.

import { useMemo } from 'react';
import { PixelRatio, Platform } from 'react-native';
import { create } from 'zustand';

export type DeviceTier = 'high' | 'mid' | 'low';

export interface DeviceTierState {
  tier: DeviceTier;
  override: DeviceTier | null;
  setOverride: (tier: DeviceTier | null) => void;
}

export const useDeviceTierStore = create<DeviceTierState>()((set) => ({
  tier: detectDefaultTier(),
  override: null,
  setOverride: (override) => set({ override }),
}));

function detectDefaultTier(): DeviceTier {
  if (Platform.OS === 'web') return 'high';
  const ratio = PixelRatio.get();
  if (ratio >= 3) return 'high';
  if (ratio >= 2) return 'mid';
  return 'low';
}

export function useDeviceTier(): {
  tier: DeviceTier;
  isHigh: boolean;
  isMid: boolean;
  isLow: boolean;
} {
  const tier = useDeviceTierStore((s) => s.override ?? s.tier);
  return useMemo(
    () => ({
      tier,
      isHigh: tier === 'high',
      isMid: tier === 'mid',
      isLow: tier === 'low',
    }),
    [tier],
  );
}

```
// Motion config that respects the OS Reduce Motion setting.
// When reduceMotion is true, `duration(ms)` returns 0 for every animation.

import { useEffect, useMemo, useState } from 'react';
import { AccessibilityInfo } from 'react-native';

export interface MotionConfig {
  readonly reduceMotion: boolean;
  readonly enabled: boolean;
  /** Returns 0 when reduceMotion is true; otherwise returns the input. */
  duration(ms: number): number;
}

export function useMotion(): MotionConfig {
  const [reduceMotion, setReduceMotion] = useState(false);

  useEffect(() => {
    let mounted = true;
    AccessibilityInfo.isReduceMotionEnabled()
      .then((enabled) => {
        if (mounted) setReduceMotion(enabled);
      })
      .catch(() => {
        // Some platforms may not implement the check; default to allowing motion.
      });

    const subscription = AccessibilityInfo.addEventListener('reduceMotionChanged', (enabled) => {
      if (mounted) setReduceMotion(enabled);
    });

    return () => {
      mounted = false;
      subscription.remove();
    };
  }, []);

  return useMemo<MotionConfig>(
    () => ({
      reduceMotion,
      enabled: !reduceMotion,
      duration: (ms: number) => (reduceMotion ? 0 : ms),
    }),
    [reduceMotion],
  );
}

// Optional haptics. Can be disabled globally.
// On Web, expo-haptics is a no-op.

import * as Haptics from 'expo-haptics';
import { useMemo } from 'react';
import { create } from 'zustand';

export interface HapticsSettings {
  enabled: boolean;
  setEnabled: (enabled: boolean) => void;
}

export const useHapticsStore = create<HapticsSettings>()((set) => ({
  enabled: true,
  setEnabled: (enabled) => set({ enabled }),
}));

export interface HapticsApi {
  readonly enabled: boolean;
  readonly setEnabled: (enabled: boolean) => void;
  light(): void;
  medium(): void;
  success(): void;
  error(): void;
}

export function useHaptics(): HapticsApi {
  const enabled = useHapticsStore((s) => s.enabled);
  const setEnabled = useHapticsStore((s) => s.setEnabled);

  return useMemo<HapticsApi>(() => {
    const safe = (fn: () => Promise<void>) => {
      if (!enabled) return;
      fn().catch(() => {
        // Swallow — haptics are non-essential and must never crash the app.
      });
    };
    return {
      enabled,
      setEnabled,
      light: () => safe(() => Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light)),
      medium: () => safe(() => Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium)),
      success: () =>
        safe(() => Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success)),
      error: () => safe(() => Haptics.notificationAsync(Haptics.NotificationFeedbackType.Error)),
    };
  }, [enabled, setEnabled]);
}

  switch (variant) {

        text: theme.colors.text.primary,
        border: 'transparent',

    case 'gold':
      return {
        bg: theme.colors.gold.fill,
        text: theme.colors.text.inverse,

        bg: theme.colors.state.danger,
        text: theme.colors.text.primary,
        border: 'transparent',
      };
    case 'icon':
      return {
        bg: theme.colors.bg.raised,
        text: theme.colors.text.primary,
        border: theme.colors.border.subtle,

  readonly variant?: TextVariant;
  /** Semantic token key (e.g. 'primary', 'accent') OR a raw color string. */
  readonly color?: TextColorKey | string;
  readonly align?: TextAlign;
  /** i18n key. When provided, `children` is ignored. */
  readonly i18nKey?: string;
  /** i18n namespace (optional; falls back to i18next default). */
  readonly ns?: string;
  /** Parameters for i18next interpolation (e.g. { count: 3 }). */
  readonly i18nParams?: Record<string, unknown>;
  readonly style?: StyleProp<TextStyle>;
  readonly children?: ReactNode;
}

export function Text({
  variant = 'body',
  color = 'primary',
  align = 'start',
  i18nKey,
  ns,
  i18nParams,
  style,
  children,
  ...rest
}: TextProps) {
  const theme = useTheme();
  const { t } = useTranslation();
  const { isRTL } = useDirection();

  const resolvedColor = useMemo<string>(() => {
    const tokens = theme.colors.text as Record<string, string>;
    return tokens[color] ?? color;
  }, [color, theme]);

  // 'auto' respects writing direction (start). 'end' needs explicit L/R swap.
  const resolvedAlign: TextStyle['textAlign'] = useMemo(() => {
    if (align === 'start') return 'auto';
    if (align === 'center') return 'center';
    return isRTL ? 'left' : 'right';
  }, [align, isRTL]);

  const content = i18nKey ? t(i18nKey, { ns, ...i18nParams }) : children;

  return (
    <RNText
      {...rest}
      style={[theme.typography[variant], { color: resolvedColor, textAlign: resolvedAlign }, style]}
    >
      {content}
    </RNText>
  );

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
  type ReactNode,
} from 'react';
import { Animated, Pressable, StyleSheet, View } from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';

import { useDirection, useMotion } from '@/hooks';
import { useTheme } from '@/theme';

import { Icon, type IconName } from './Icon';
import { Text } from './Text';

export type ToastVariant = 'success' | 'error' | 'info' | 'warning';

export interface ToastOptions {
  /** Pre-translated message. Caller uses t('...') before calling show(). */
  readonly message: string;
  readonly variant?: ToastVariant;
  /** Auto-dismiss duration in ms. Default 3000. 0 disables auto-dismiss. */
  readonly duration?: number;
}

interface ToastInternal extends ToastOptions {
  readonly id: number;
}

export interface ToastContextValue {
  show(options: ToastOptions): void;
  hide(): void;
}

const ToastContext = createContext<ToastContextValue | null>(null);

export interface ToastProviderProps {
  readonly children: ReactNode;
}

export function ToastProvider({ children }: ToastProviderProps) {
  const theme = useTheme();
  const { isRTL } = useDirection();
  const motion = useMotion();
  const insets = useSafeAreaInsets();

  const [toast, setToast] = useState<ToastInternal | null>(null);
  const [opacity] = useState(() => new Animated.Value(0));
  const [translateY] = useState(() => new Animated.Value(-20));
  const idCounter = useRef(0);
  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const hide = useCallback(() => {
    const dur = motion.duration(200);
    if (dur === 0) {
      setToast(null);
      return;
    }
    Animated.parallel([
      Animated.timing(opacity, { toValue: 0, duration: dur, useNativeDriver: true }),
      Animated.timing(translateY, { toValue: -20, duration: dur, useNativeDriver: true }),
    ]).start(() => setToast(null));
  }, [motion, opacity, translateY]);

  const show = useCallback(
    (options: ToastOptions) => {
      if (timerRef.current) {
        clearTimeout(timerRef.current);
        timerRef.current = null;
      }
      idCounter.current += 1;
      setToast({ ...options, id: idCounter.current });

      const enterDur = motion.duration(220);
      if (enterDur === 0) {
        opacity.setValue(1);
        translateY.setValue(0);
      } else {
        opacity.setValue(0);
        translateY.setValue(-20);
        Animated.parallel([
          Animated.timing(opacity, { toValue: 1, duration: enterDur, useNativeDriver: true }),
          Animated.timing(translateY, { toValue: 0, duration: enterDur, useNativeDriver: true }),
        ]).start();
      }

      const duration = options.duration ?? 3000;
      if (duration > 0) {
        timerRef.current = setTimeout(() => hide(), duration);
      }
    },
    [hide, motion, opacity, translateY],
  );

  useEffect(() => {
    return () => {
      if (timerRef.current) clearTimeout(timerRef.current);
    };
  }, []);

  const value = useMemo<ToastContextValue>(() => ({ show, hide }), [hide, show]);

  const variant = toast?.variant ?? 'info';
  const iconName: IconName =
    variant === 'success'
      ? 'checkCircle'
      : variant === 'error'
        ? 'xCircle'
        : variant === 'warning'
          ? 'alertTriangle'
          : 'info';
  const accentColor =
    variant === 'success'
      ? theme.colors.state.success
      : variant === 'error'
        ? theme.colors.state.danger
        : variant === 'warning'
          ? theme.colors.gold.fill
          : theme.colors.state.info;

  return (
    <ToastContext.Provider value={value}>
      <View style={styles.root}>
        {children}
        {toast ? (
          <Animated.View
            pointerEvents="box-none"
            style={[
              styles.container,
              {
                paddingTop: insets.top + theme.spacing[3],
                paddingStart: theme.spacing[4],
                paddingEnd: theme.spacing[4],
                opacity,
                transform: [{ translateY }],
              },
            ]}
          >
            <View
              accessibilityRole="alert"
              accessibilityLiveRegion="polite"
              style={[
                styles.toast,
                {
                  backgroundColor: theme.colors.bg.raised,
                  borderColor: accentColor,
                  borderRadius: theme.radius.md,
                  paddingStart: theme.spacing[3],
                  paddingEnd: theme.spacing[3],
                  flexDirection: isRTL ? 'row-reverse' : 'row',
                },
              ]}
              testID="toast"
            >
              <Icon name={iconName} size={20} color={accentColor} />
              <Text variant="body" color="primary" style={styles.message}>
                {toast.message}
              </Text>
              <Pressable
                onPress={hide}
                hitSlop={8}
                accessibilityRole="button"
                accessibilityLabel="dismiss"
                testID="toast-dismiss"
              >
                <Icon name="x" size={18} color={theme.colors.text.secondary} />
              </Pressable>
            </View>
          </Animated.View>
        ) : null}
      </View>
    </ToastContext.Provider>
  );
}

  PanResponder,
  Pressable,
  StyleSheet,
  View,
  useWindowDimensions,
} from 'react-native';
import { useSafeAreaInsets } from 'react-native-safe-area-context';

import { useMotion } from '@/hooks';
import { useTheme } from '@/theme';

export interface BottomSheetProps {
  readonly visible: boolean;
  readonly onClose: () => void;
  readonly children: ReactNode;
  readonly dismissOnBackdrop?: boolean;
  readonly showHandle?: boolean;
  /** Max height as percent of the screen (0-100). Default 85. */
  readonly maxHeightPercent?: number;
  readonly testID?: string;
}

/**
 * Modal bottom sheet. Single snap (maxHeightPercent of screen).
 * Drag down to dismiss (threshold 100px or vy > 0.5).
 * Do NOT open a Dialog and a BottomSheet at the same time.
 */
export function BottomSheet({
  visible,
  onClose,
  children,
  dismissOnBackdrop = true,
  showHandle = true,
  maxHeightPercent = 85,
  testID,
}: BottomSheetProps) {
  const theme = useTheme();
  const motion = useMotion();
  const insets = useSafeAreaInsets();
  const { height: screenHeight } = useWindowDimensions();

  const sheetHeight = (screenHeight * maxHeightPercent) / 100;

  const [translateY] = useState(() => new Animated.Value(sheetHeight));
  const [opacity] = useState(() => new Animated.Value(0));

  useEffect(() => {
    if (!visible) return;
    const dur = motion.duration(220);
    if (dur === 0) {
      translateY.setValue(0);
      opacity.setValue(1);
      return;
    }
    translateY.setValue(sheetHeight);
    opacity.setValue(0);
    Animated.parallel([
      Animated.timing(translateY, { toValue: 0, duration: dur, useNativeDriver: true }),
      Animated.timing(opacity, { toValue: 1, duration: dur, useNativeDriver: true }),
    ]).start();
  }, [visible, motion, translateY, opacity, sheetHeight]);

  const close = () => {
    const dur = motion.duration(200);
    if (dur === 0) {
      onClose();
      return;
    }
    Animated.parallel([
      Animated.timing(translateY, {
        toValue: sheetHeight,
        duration: dur,
        useNativeDriver: true,
      }),
      Animated.timing(opacity, { toValue: 0, duration: dur, useNativeDriver: true }),
    ]).start(() => onClose());
  };

  const panResponder = useMemo(
    () =>
      PanResponder.create({
        onStartShouldSetPanResponder: () => false,
        onMoveShouldSetPanResponder: (_e, g) => Math.abs(g.dy) > 6 && g.dy > 0,
        onPanResponderMove: (_e, g) => {
          if (g.dy > 0) translateY.setValue(g.dy);
        },
        onPanResponderRelease: (_e, g) => {
          if (g.dy > 100 || g.vy > 0.5) {
            close();
          } else {
            Animated.spring(translateY, {
              toValue: 0,
              useNativeDriver: true,
              damping: 20,
              stiffness: 200,
            }).start();
          }
        },
      }),
    // close is stable enough for the lifetime of the component.
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [translateY, sheetHeight],
  );

  return (
    <Modal
      visible={visible}
      transparent
      animationType="none"
      onRequestClose={onClose}
      statusBarTranslucent
    >
      <Animated.View
        style={[styles.backdrop, { backgroundColor: theme.colors.bg.overlay, opacity }]}
      >
        <Pressable
          style={StyleSheet.absoluteFill}
          onPress={() => {
            if (dismissOnBackdrop) close();
          }}
          accessibilityRole="button"
          accessibilityLabel="close"
          testID={testID ? `${testID}-backdrop` : undefined}
        />
        <Animated.View
          accessibilityViewIsModal
          {...panResponder.panHandlers}
          style={[
            styles.sheet,
            {
              backgroundColor: theme.colors.bg.raised,
              borderTopStartRadius: theme.radius.xl,
              borderTopEndRadius: theme.radius.xl,
              height: sheetHeight,
              paddingBottom: insets.bottom + theme.spacing[4],
              transform: [{ translateY }],
            },
          ]}
          testID={testID}
        >
          {showHandle ? (
            <View style={styles.handleRow} pointerEvents="none">
              <View style={[styles.handle, { backgroundColor: theme.colors.border.strong }]} />
            </View>
          ) : null}
          <View style={{ flex: 1, paddingHorizontal: theme.spacing[4] }}>{children}</View>
        </Animated.View>
      </Animated.View>
    </Modal>
  );
}

const styles = StyleSheet.create({
  backdrop: { flex: 1, justifyContent: 'flex-end' },
  sheet: {
    width: '100%',
    alignSelf: 'stretch',
    overflow: 'hidden',
  },
  handleRow: { alignItems: 'center', paddingVertical: 10 },
  handle: { width: 40, height: 4, borderRadius: 2 },
});

import { StyleSheet, View, type StyleProp, type ViewStyle } from 'react-native';
import Animated, {
  cancelAnimation,
  useAnimatedStyle,
  useSharedValue,
  withRepeat,
  withTiming,
} from 'react-native-reanimated';

import { useMotion } from '@/hooks';
import { useTheme } from '@/theme';

import { Text } from './Text';

export type ProgressBarSize = 'sm' | 'md' | 'lg';
export type ProgressBarColor = 'accent' | 'gold' | 'success' | 'danger';

const HEIGHTS: Record<ProgressBarSize, number> = {
  sm: 4,
  md: 8,
  lg: 12,
};

export interface ProgressBarProps {
  /** 0..1. Omit for indeterminate. */
  readonly value?: number;
  readonly size?: ProgressBarSize;
  readonly color?: ProgressBarColor;
  /** Show a percentage label to the end of the bar. Determinate only. */
  readonly showLabel?: boolean;
  readonly style?: StyleProp<ViewStyle>;
  readonly testID?: string;
}

export function ProgressBar({
  value,
  size = 'md',
  color = 'accent',
  showLabel = false,
  style,
  testID,
}: ProgressBarProps) {
  const theme = useTheme();
  const motion = useMotion();
  const height = HEIGHTS[size];
  const isIndeterminate = value === undefined;

  const resolvedColor =
    color === 'gold'
      ? theme.colors.gold.fill
      : color === 'success'
        ? theme.colors.state.success
        : color === 'danger'
          ? theme.colors.state.danger
          : theme.colors.accent.fill;

  const clamped = isIndeterminate ? 0 : Math.min(1, Math.max(0, value!));
  const percent = Math.round(clamped * 100);

  const opacity = useSharedValue(1);

  useEffect(() => {
    if (!isIndeterminate) {
      opacity.value = 1;
      return;
    }
    if (motion.reduceMotion) {
      opacity.value = 0.7;
      return;
    }
    opacity.value = 0.4;
    opacity.value = withRepeat(withTiming(0.9, { duration: 900 }), -1, true);
    return () => {
      cancelAnimation(opacity);
    };
  }, [isIndeterminate, motion.reduceMotion, opacity]);

  const animatedStyle = useAnimatedStyle(() => ({ opacity: opacity.value }));

  return (
    <View style={[styles.container, style]} testID={testID}>
      <View
        accessible={true}
        accessibilityRole="progressbar"
        accessibilityValue={
          isIndeterminate
            ? undefined
            : { min: 0, max: 100, now: percent }
        }

import { useEffect, useState } from 'react';
import {
  Animated,
  Pressable,
  StyleSheet,
  View,
  type LayoutChangeEvent,
} from 'react-native';

import { useDirection, useHaptics, useMotion } from '@/hooks';
import { useTheme } from '@/theme';

import { Icon, type IconName } from './Icon';
import { Text } from './Text';

export interface SegmentOption<V extends string> {
  readonly value: V;
  readonly label: string;
  readonly icon?: IconName;
  readonly disabled?: boolean;
}

export type SegmentedSize = 'sm' | 'md';

const SIZES: Record<SegmentedSize, { height: number; font: 'caption' | 'bodyStrong' }> = {
  sm: { height: 36, font: 'caption' },
  md: { height: 44, font: 'bodyStrong' },
};

export interface SegmentedControlProps<V extends string> {
  readonly segments: readonly SegmentOption<V>[];
  readonly value: V;
  readonly onChange: (value: V) => void;
  readonly size?: SegmentedSize;
  readonly testID?: string;
}

/**
 * Material-style segmented control with animated indicator.
 * Equal-width segments. RTL-aware via `start: 0` + translateX flip.
 */
export function SegmentedControl<V extends string>({
  segments,
  value,
  onChange,
  size = 'md',
  testID,
}: SegmentedControlProps<V>) {
  const theme = useTheme();
  const { isRTL } = useDirection();
  const motion = useMotion();
  const haptics = useHaptics();
  const spec = SIZES[size];

  const [containerWidth, setContainerWidth] = useState(0);
  const segmentWidth = segments.length > 0 ? containerWidth / segments.length : 0;
  const selectedIndex = Math.max(
    0,
    segments.findIndex((s) => s.value === value),
  );

  const [translateX] = useState(() => new Animated.Value(0));

  useEffect(() => {
    if (segmentWidth === 0) return;
    const target = selectedIndex * segmentWidth * (isRTL ? -1 : 1);
    const dur = motion.duration(220);
    if (dur === 0) {
      translateX.setValue(target);
    } else {
      Animated.timing(translateX, {
        toValue: target,
        duration: dur,
        useNativeDriver: true,
      }).start();
    }
  }, [selectedIndex, segmentWidth, isRTL, motion, translateX]);

  const onLayout = (e: LayoutChangeEvent) => {
    const w = e.nativeEvent.layout.width;
    if (w !== containerWidth) setContainerWidth(w);
  };

  return (
    <View
      onLayout={onLayout}
      accessibilityRole="tablist"
      style={[
        styles.container,
        {
          height: spec.height,
          backgroundColor: theme.colors.bg.surface,
          borderRadius: theme.radius.full,
          borderWidth: 1,
          borderColor: theme.colors.border.subtle,
          flexDirection: isRTL ? 'row-reverse' : 'row',
          padding: 3,
        },
      ]}
      testID={testID}
    >
      {containerWidth > 0 && segmentWidth > 0 ? (
        <Animated.View
          pointerEvents="none"
          style={[
            styles.indicator,
            {
              width: segmentWidth,
              height: spec.height - 6,
              borderRadius: theme.radius.full,
              backgroundColor: theme.colors.accent.fill,
              start: 3,
              transform: [{ translateX }],
            },
          ]}
        />
      ) : null}

      {segments.map((seg) => {
        const selected = seg.value === value;
        const isDisabled = !!seg.disabled;
        const textColor = selected ? 'primary' : isDisabled ? 'tertiary' : 'secondary';
        return (
          <Pressable
            key={seg.value}
            accessibilityRole="tab"
            accessibilityState={{ selected, disabled: isDisabled }}
            accessibilityLabel={seg.label}
            disabled={isDisabled}
            onPress={() => {
              if (isDisabled) return;
              haptics.light();
              onChange(seg.value);
            }}
            style={styles.segment}
          >
            <View
              style={[
                styles.segmentContent,
                { flexDirection: isRTL ? 'row-reverse' : 'row', gap: 6 },
              ]}
            >
              {seg.icon ? (
                <Icon name={seg.icon} size={14} color={theme.colors.text[textColor]} />
              ) : null}
              <Text variant={spec.font} color={textColor} numberOfLines={1}>
                {seg.label}
              </Text>
            </View>
          </Pressable>
        );
      })}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    position: 'relative',
    alignItems: 'center',
  },
  indicator: {
    position: 'absolute',
    top: 3,
  },
  segment: {
    flex: 1,
    alignItems: 'center',
    justifyContent: 'center',
    minHeight: 32,
  },
  segmentContent: { alignItems: 'center' },
});

import { useEffect, useState } from 'react';
import { Animated, Pressable, StyleSheet, View } from 'react-native';

import { useDirection, useHaptics, useMotion } from '@/hooks';
import { useTheme } from '@/theme';

import { Text } from '../Text';

export type SwitchSize = 'sm' | 'md';

const SIZES: Record<SwitchSize, { trackW: number; trackH: number; thumb: number; inset: number }> = {
  sm: { trackW: 44, trackH: 24, thumb: 18, inset: 3 },
  md: { trackW: 52, trackH: 30, thumb: 24, inset: 3 },
};

export interface SwitchProps {
  readonly value: boolean;
  readonly onValueChange: (value: boolean) => void;
  readonly disabled?: boolean;
  readonly size?: SwitchSize;
  readonly label?: string;
  /** Place the label before the switch (start side). Default false (after). */
  readonly labelFirst?: boolean;
  readonly testID?: string;
}

export function Switch({
  value,
  onValueChange,
  disabled = false,
  size = 'md',
  label,
  labelFirst = false,
  testID,
}: SwitchProps) {
  const theme = useTheme();
  const { isRTL } = useDirection();
  const motion = useMotion();
  const haptics = useHaptics();
  const spec = SIZES[size];

  const travel = spec.trackW - spec.thumb - spec.inset * 2;
  const [anim] = useState(() => new Animated.Value(value ? 1 : 0));

  useEffect(() => {
    const toValue = value ? 1 : 0;
    const duration = motion.duration(180);
    if (duration === 0) {
      anim.setValue(toValue);
    } else {
      Animated.timing(anim, {
        toValue,
        duration,
        useNativeDriver: true,
      }).start();
    }
  }, [value, anim, motion]);

  const translateX = anim.interpolate({
    inputRange: [0, 1],
    outputRange: isRTL
      ? [spec.inset + travel, spec.inset]
      : [spec.inset, spec.inset + travel],
  });

  const trackBg = value ? theme.colors.accent.fill : theme.colors.bg.raised;
  const trackBorder = value ? theme.colors.accent.fill : theme.colors.border.subtle;

  const onPress = () => {
    if (disabled) return;
    haptics.light();
    onValueChange(!value);
  };

  const control = (
    <Pressable
      accessibilityRole="switch"
      accessibilityState={{ checked: value, disabled }}
      accessibilityLabel={label}
      onPress={onPress}
      disabled={disabled}
      testID={testID}
      hitSlop={6}
      style={{ opacity: disabled ? 0.45 : 1 }}
    >
      <View
        style={[
          styles.track,
          {
            width: spec.trackW,
            height: spec.trackH,
            borderRadius: spec.trackH / 2,
            backgroundColor: trackBg,
            borderColor: trackBorder,
          },
        ]}
      >
        <Animated.View
          style={[
            styles.thumb,
            {
              width: spec.thumb,
              height: spec.thumb,
              borderRadius: spec.thumb / 2,
              backgroundColor: theme.colors.text.primary,
              transform: [{ translateX }],
            },
          ]}
        />
      </View>
    </Pressable>
  );

  if (!label) return control;

  return (
    <View style={styles.row}>
      {labelFirst ? (
        <>
          <Pressable onPress={onPress} disabled={disabled} accessibilityElementsHidden>
            <Text variant="body" color={disabled ? 'tertiary' : 'primary'}>
              {label}
            </Text>
          </Pressable>
          {control}
        </>
      ) : (
        <>
          {control}
          <Pressable onPress={onPress} disabled={disabled} accessibilityElementsHidden>
            <Text variant="body" color={disabled ? 'tertiary' : 'primary'}>
              {label}
            </Text>
          </Pressable>
        </>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  track: {
    borderWidth: 1,
    justifyContent: 'center',
  },
  thumb: {
    position: 'absolute',
  },
  row: {
    flexDirection: 'row',
    alignItems: 'center',
    gap: 12,
  },
});

  readonly progress: SharedValue<number>;
  readonly background?: ReactNode;
  readonly midground?: ReactNode;
  readonly foreground: ReactNode;
  readonly height?: number;
  readonly testID?: string;

  midground,
  foreground,

// Uses Reanimated SharedValue + withRepeat. When reduceMotion is true,
// the rings stay static (still visible, no motion).

import { useEffect } from 'react';
import { View } from 'react-native';
import Animated, {
  cancelAnimation,
  useAnimatedStyle,
  useSharedValue,
  withRepeat,
  withTiming,
} from 'react-native-reanimated';
import Svg, { Circle } from 'react-native-svg';

import { useMotion } from '@/hooks';
import { useTheme } from '@/theme';

export interface DoubleRingLoaderProps {
  readonly size?: number;
  readonly color?: string;
  readonly testID?: string;
  readonly accessibilityLabel?: string;
}

export function DoubleRingLoader({
  size = 64,
  color,
  testID,
  accessibilityLabel = 'loading',
}: DoubleRingLoaderProps) {
  const theme = useTheme();
  const motion = useMotion();
  const stroke = color ?? theme.colors.accent.fill;
  const rotation = useSharedValue(0);

  useEffect(() => {
    if (motion.reduceMotion) {
      rotation.value = 0;
      return;
    }
    rotation.value = withRepeat(withTiming(360, { duration: 1400 }), -1, false);
    return () => {
      cancelAnimation(rotation);
    };
  }, [motion.reduceMotion, rotation]);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ rotate: `${rotation.value}deg` }],
  }));

  const strokeWidth = Math.max(2, size * 0.06);
  const outerR = size / 2 - strokeWidth / 2 - 2;
  const innerR = outerR * 0.62;
  const cx = size / 2;
  const cy = size / 2;
  const ringOffset = outerR * 0.35;

  const outerDash = 2 * Math.PI * outerR * 0.72;
  const innerDash = 2 * Math.PI * innerR * 0.62;

  return (
    <View
      accessibilityRole="progressbar"
      accessibilityLabel={accessibilityLabel}
      testID={testID}
    >
      <Animated.View style={[{ width: size, height: size }, animatedStyle]}>
        <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
          {/* Outer / left ring */}
          <Circle
            cx={cx - ringOffset}
            cy={cy}
            r={outerR}
            stroke={stroke}
            strokeWidth={strokeWidth}
            strokeLinecap="round"
            strokeDasharray={`${outerDash} ${2 * Math.PI * outerR}`}
            fill="none"
          />
          {/* Inner / right ring (interlocks with outer) */}
          <Circle
            cx={cx + ringOffset}
            cy={cy}
            r={innerR}
            stroke={stroke}
            strokeWidth={strokeWidth}
            strokeLinecap="round"
            strokeDasharray={`${innerDash} ${2 * Math.PI * innerR}`}
            fill="none"
          />

 * ويُستخدم فقط لمحتوى غير-UI (مثل fixture في اختبار). ممنوع لنصوص الواجهة.
 */
import { readdir, readFile } from 'node:fs/promises';
import { join, extname, relative } from 'node:path';

const ROOT = process.cwd();
const SCAN_ROOTS = ['app', 'src'];
const SKIP_DIR_NAMES = new Set([
  'node_modules',
  '.git',
  '.expo',
  'dist',
  'web-build',
  'build',
  'coverage',
]);
const ALLOW_PATH_PREFIXES = [
  'src/locales/', // المكان الوحيد المسموح فيه بالنصوص العربية
];
const INLINE_ALLOW = 'i18n-allow-arabic';

// Arabic blocks:
//   0600–06FF  Arabic
//   0750–077F  Arabic Supplement
//   08A0–08FF  Arabic Extended-A
//   FB50–FDFF  Arabic Presentation Forms-A
//   FE70–FEFF  Arabic Presentation Forms-B
const ARABIC_RE = /[\u0600-\u06FF\u0750-\u077F\u08A0-\u08FF\uFB50-\uFDFF\uFE70-\uFEFF]/;

const findings = [];

async function walk(dir) {
  let entries;
  try {
    entries = await readdir(dir, { withFileTypes: true });
  } catch {
    return; // المجلد غير موجود — لا مشكلة
  }
  for (const entry of entries) {
    const fullPath = join(dir, entry.name);
    if (entry.isDirectory()) {
      if (SKIP_DIR_NAMES.has(entry.name)) continue;
      await walk(fullPath);
      continue;
    }
    if (!entry.isFile()) continue;
    const rel = relative(ROOT, fullPath).split('\\').join('/');
    if (ALLOW_PATH_PREFIXES.some((prefix) => rel.startsWith(prefix))) continue;
    const ext = extname(fullPath);
    if (ext !== '.ts' && ext !== '.tsx') continue;
    const content = await readFile(fullPath, 'utf8');
    const lines = content.split('\n');
    let inBlockComment = false; // متغير لتتبع التعليقات الكتلية متعددة الأسطر

    lines.forEach((line, idx) => {
      if (line.includes(INLINE_ALLOW)) return;

      let codeToCheck = line;

      // 1. التعامل مع التعليقات الكتلية (/* ... */)
      if (inBlockComment) {
        const endIdx = codeToCheck.indexOf('*/');
        if (endIdx !== -1) {
          codeToCheck = codeToCheck.slice(endIdx + 2); // احتفظ بما بعد إغلاق التعليق
          inBlockComment = false;
        } else {
          return; // السطر بالكامل تعليق، تجاهله
        }
      }

      const startIdx = codeToCheck.indexOf('/*');
      if (startIdx !== -1) {
        const endIdx = codeToCheck.indexOf('*/', startIdx + 2);
        if (endIdx !== -1) {
          // التعليق يبدأ وينتهي في نفس السطر
          codeToCheck = codeToCheck.slice(0, startIdx) + codeToCheck.slice(endIdx + 2);
        } else {
          // التعليق يبدأ هنا ويستمر للأسطر التالية
          codeToCheck = codeToCheck.slice(0, startIdx);
          inBlockComment = true;
        }
      }

      // 2. التعامل مع تعليقات السطر الواحد (//)
      const lineCommentIdx = codeToCheck.indexOf('//');
      if (lineCommentIdx !== -1) {
        codeToCheck = codeToCheck.slice(0, lineCommentIdx);
      }

      // 3. الفحص النهائي (نفحص النص الخالي من التعليقات فقط)
      if (ARABIC_RE.test(codeToCheck)) {
        findings.push({ rel, line: idx + 1, text: line.trim().slice(0, 160) });
      }
    });
  }
}

for (const root of SCAN_ROOTS) {
  await walk(join(ROOT, root));
}

if (findings.length > 0) {
  console.error('✗ Hardcoded Arabic detected in non-locale TypeScript files:\n');
  for (const f of findings) {
    console.error(`  ${f.rel}:${f.line}: ${f.text}`);
  }
  console.error(`\nFound ${findings.length} occurrence(s).`);
  console.error('Move these strings to src/locales/*.json and reference them via i18n keys.');
  console.error(`If a line is legitimately not UI (e.g. a test fixture), append the marker: ${INLINE_ALLOW}`);
  process.exit(1);
}

console.log('✓ No hardcoded Arabic found outside src/locales.');
```

// populateDemoSeed — injects demo data into the mock repositories.
// Honest about coverage: some repositories do not expose create/update
// via their contract (content/notification/search). Those are reported
// as `unsupported` rather than silently skipped.
//
// Do NOT call this in backend mode. It intentionally no-ops and reports
// `skipped: 'not-demo-mode'`.

import type { RepositoryContainer } from '@/repositories/container';

import { buildDemoSeedBundle } from './index';

export interface PopulateReport {
  readonly ok: boolean;
  readonly skipped: 'not-demo-mode' | null;
  readonly created: Readonly<Record<string, number>>;
  readonly unsupported: readonly string[];
  readonly errors: readonly string[];
}

const UNSUPPORTED_REPOS = ['content', 'notifications', 'search'] as const;

export async function populateDemoSeed(
  container: RepositoryContainer,
): Promise<PopulateReport> {
  if (container.mode !== 'demo') {
    return {
      ok: true,
      skipped: 'not-demo-mode',
      created: {},
      unsupported: [],
      errors: [],
    };
  }

  const bundle = buildDemoSeedBundle();
  const created: Record<string, number> = {};
  const errors: string[] = [];

  async function tryCreate(
    name: string,
    items: readonly unknown[],
    fn: (item: unknown) => Promise<{ ok: boolean; error?: unknown }>,
  ): Promise<void> {
    let count = 0;
    for (const item of items) {
      const r = await fn(item);
      if (r.ok) {
        count += 1;
      } else {
        errors.push(`${name}: ${JSON.stringify(r.error)}`);
      }
    }
    created[name] = count;
  }

  await tryCreate('profiles', bundle.profiles, (i) =>
    container.profiles.create(i as (typeof bundle.profiles)[number]),
  );
  await tryCreate('programs', bundle.programs, (i) =>
    container.programs.create(i as (typeof bundle.programs)[number]),
  );
  await tryCreate('applications', bundle.applications, (i) =>
    container.applications.create(i as (typeof bundle.applications)[number]),
  );
  await tryCreate('weddings', bundle.weddings, (i) =>
    container.weddings.create(i as (typeof bundle.weddings)[number]),
  );
  await tryCreate('partners', bundle.partners, (i) =>
    container.partners.create(i as (typeof bundle.partners)[number]),
  );
  await tryCreate('offers', bundle.offers, (i) =>
    container.offers.create(i as (typeof bundle.offers)[number]),
  );
  await tryCreate('souk', bundle.souk, (i) =>
    container.souk.create(i as (typeof bundle.souk)[number]),
  );
  await tryCreate('jobs', bundle.jobs, (i) =>
    container.jobs.create({
      titleKey: (i as (typeof bundle.jobs)[number]).titleKey,
      partnerId: (i as (typeof bundle.jobs)[number]).partnerId,
    }),
  );
  await tryCreate('volunteers', bundle.volunteers, (i) =>
    container.volunteers.create(i as (typeof bundle.volunteers)[number]),
  );
  await tryCreate('donations', bundle.donations, (i) =>
    container.donations.create(i as (typeof bundle.donations)[number]),
  );
  await tryCreate('disbursements', bundle.disbursements, (i) =>
    container.disbursements.create(i as (typeof bundle.disbursements)[number]),
  );
  await tryCreate('audit', bundle.audit, (i) =>
    container.audit.append(i as (typeof bundle.audit)[number]),
  );

  // Settings uses update(), not create().
  let settingsCount = 0;
  for (const s of bundle.settings) {
    const r = await container.settings.update(s.key, s.value);
    if (r.ok) settingsCount += 1;
    else errors.push(`settings: ${JSON.stringify(r.error)}`);
  }
  created.settings = settingsCount;

  return {
    ok: errors.length === 0,
    skipped: null,
    created: Object.freeze(created),
    unsupported: [...UNSUPPORTED_REPOS],
    errors: Object.freeze(errors),
  };
}

## ADR-0008 — eslint-import-resolver-typescript لحل مسارات `@/`

- **التاريخ:** 2026-11-09

## ADR-0011 — Jest mock لـreact-native-reanimated

- **التاريخ:** 2026-11-14
- **الحالة:** Accepted
- **السياق:** P04.5b هو أول استخدام لـ`react-native-reanimated` في المشروع. `jest-expo` preset لا يفعّل `react-native-worklets/plugin` في Babel/Jest، لذا أي `import` من `react-native-reanimated` في اختبار يفشل بخطأ "worklet not compiled". لا يوجد mock رسمي مضمّن لـReanimated 4 في `jest-expo`.
- **القرار:** إضافة mock **يدوي كامل** في `jest.setup.ts` يُعيد:
  - `Animated.View/Text/ScrollView/Image` → مكونات RN عادية.
  - `useSharedValue(v)` → `{ value: v }` (كائن عادي، ليس ref).
  - `useAnimatedStyle(fn)` → `fn()` (تنفيذ مرة واحدة عند الرندر).
  - `withTiming/withSpring/withRepeat/etc.` → إرجاع القيمة النهائية مباشرة.
  - `useAnimatedRef` → `{ current: null }`.
  - `Gesture.*` + `GestureDetector` → كائنات chainable no-op.
- **البدائل المدروسة:**
  - `require('react-native-reanimated').setUpTests()` — يفشل مع Reanimated 4 في `jest-expo`.
  - `react-native-reanimated/mock` — مسار قديم غير مضمون في v4.
  - عدم استخدام Reanimated أصلاً — يخالف brief §8 "الحركة عبر Worklets".
- **التبعات:**
  - + كل اختبارات المكونات المتحركة تعمل بثبات (487 + ~20 اختبار P04.5b).
  - − الحركة الفعلية **لا تُختبر** في Jest. تُختبر على الأجهزة الفعلية + E2E في P21 (Playwright).
  - − إذا احتاجت أي ميزة قياس قيم متوسطة من `SharedValue` (مثلاً، قيمة أثناء الحركة)، Jest لن يعكسها.
  - − ترقيات Reanimated المستقبلية (v5) قد تكسر هذا الـmock. موثَّق للصيانة.
- **معايير المراجعة:** عندما يوفّر `jest-expo` رسمياً mock لـReanimated 4، أو عندما يصبح `setUpTests()` يعمل بثبات. عندها يُزال هذا الـmock.
- **المراجع:** `jest.setup.ts`, `src/components/brand/DoubleRingLoader.tsx`, `src/components/brand/Hero3D.tsx`.

## 3) Rules

### 3.1 Global Do/Don't

**Do:**
- Use semantic tokens from `useTheme()` — never hex literals in components.
- Pass i18n keys through `i18nKey` prop on `<Text>`, or pre-translate with `t()` before passing `label`/`title`/`message`.
- Use logical properties (`paddingStart/End`, `marginStart/End`, `start`/`end`) — never `left`/`right`.
- Test on Tier low: no glows, no heavy motion, no blur.
- Ensure every interactive element has `accessibilityRole` + accessible label.
- Ensure every interactive target is ≥44dp (`MIN_TOUCH_TARGET`).

**Don't:**
- Hardcode Arabic (or any) UI strings in components — put them in `src/locales/*.json`.
- Stack `<Dialog>` and `<BottomSheet>` simultaneously.
- Use `<GlowView>` for decorative purposes — only for hero emphasis.
- Use `<GlassView>` outside Dock and Sheet contexts.
- Use `text.accent` on `bg.raised` — use `text.accentOnRaised` instead.
- Rely on color alone to convey state (always pair with icon or text).
- Use `prettier` skip comments to avoid reformatting — run `format` instead.

### 3.2 Glow

**Maximum 1 glow per screen. 2 only for ceremonial hero (e.g., onboarding completion).**
- Button `primary` / `gold` variants apply glow automatically on Tier high/mid — do not double-wrap in `<GlowView>`.
- Use `<GlowView>` only when the button's own glow is insufficient (e.g., wrapping an illustration or card).
- On Tier low, glows are disabled. **Never** compensate with stronger shadows — the intent is a lighter surface for slow devices.

### 3.3 Glass

**Only for Dock + Sheet.**
- `<GlassView>` uses `expo-blur` on Tier high/mid, and a solid `rgba(30,24,50,0.92)` fallback on Tier low.
- Do not use `<GlassView>` for lists, forms, or cards — they should be on `bg.surface` or `bg.raised` for readability.
- On Web, uses `backdrop-filter` — verify visibility on older browsers before shipping.

### 3.4 Motion

- **Every animation duration must pass through `useMotion().duration(ms)`.** This ensures Reduce Motion is respected.
- Do not introduce an animation that changes the layout of `Pressable` targets (breaks focus).
- Skeleton / ProgressBar indeterminate pulses are default-on; skip on Tier low via `useDeviceTier`.
- Reanimated worklets compile via `babel-preset-expo` (SDK 58) — no config needed. Test in Jest uses a static mock (ADR-0011).

### 3.5 RTL Checklist

Every component must satisfy all of these before merge:
- [ ] `flexDirection: isRTL ? 'row-reverse' : 'row'` on horizontal layouts.
- [ ] `paddingStart`/`paddingEnd` — never `paddingLeft`/`paddingRight`.
- [ ] `marginStart`/`marginEnd` — never `marginLeft`/`marginRight`.
- [ ] Directional icons (`chevronLeft`, `chevronRight`, `arrowLeft`, `arrowRight`) pass through `<Icon>`'s auto-mirror.
- [ ] `textAlign: 'auto'` for start-aligned text.
- [ ] Numeric inputs (`keyboardType: 'numeric'|'phone-pad'|'email-address'`) force `textAlign: 'left'` + `writingDirection: 'ltr'`.
- [ ] Tab / Segment indicator uses `start: 0` + translated `translateX` (flipped for RTL).
- [ ] Any `Absolute` positioning uses `start`/`end` (not `left`/`right`).

### 3.6 Accessibility Checklist

Every component must satisfy all of these before merge:
- [ ] `accessibilityRole` set on interactive elements (`button`, `link`, `checkbox`, `switch`, `radio`, `tab`, `progressbar`, `alert`, `image`, `header`).
- [ ] `accessibilityLabel` set on icon-only buttons and complex images.
- [ ] `accessibilityState` reflects real state (`{ checked, disabled, selected, busy, expanded }`).
- [ ] `accessibilityValue` for progress elements (`{ min, max, now }`).
- [ ] `accessibilityViewIsModal` on Dialog and BottomSheet.
- [ ] Decorative elements (Divider, Skeleton) are hidden via `accessibilityElementsHidden` + `importantForAccessibility="no-hide-descendants"`.
- [ ] Every interactive target ≥44dp (`hitSlop` when needed).
- [ ] Focus states visible on Web (`Button`, `TextInput`, Tabs).
- [ ] No text conveyed only via color (pair with icon or label).
- [ ] Contrast: all text passes WCAG AA (4.5:1 for body, 3:1 for large ≥18pt).

### 3.7 Tier Matrix

| Component | High | Mid | Low |
|-----------|------|-----|-----|
| Button (primary/gold) glow | ✅ | ✅ | ❌ |
| GlowView | ✅ | ✅ | ❌ (flat) |
| GlassView | ✅ blur | ✅ blur | ✅ solid fallback |
| Hero3D parallax | ✅ | ✅ | ❌ (flat layers) |
| Skeleton shimmer | ✅ | ✅ | ❌ (static 0.5) |
| ProgressBar indeterminate pulse | ✅ | ✅ | ❌ (static 0.7) |
| DoubleRingLoader / Success | ✅ | ✅ | ✅ (respects reduceMotion, not tier) |
| QamariyaPattern opacity | ✅ | ✅ | ✅ (SVG, low cost) |

### 3.8 Performance Notes

- **SVG-heavy components** (`QamariyaPattern`, `ArchHero`, `RingProgress`, `DoubleRing*`) render cheaply on all tiers.
- **Blur** (`GlassView`) is the most expensive surface — restricted to Dock + Sheet, disabled on Tier low.
- **Reanimated worklets** used in `DoubleRing*`, `Hero3D`, `Skeleton`, `ProgressBar` — small, focused, driven by SharedValues. No layout animations.
- **Gallery** lives at `/gallery` — excluded from production bundle? **No.** Deferred to P22 (development-only gating).
- **Bundle size**: `lucide-react-native` is tree-shaken by Metro; only 39 icons imported. `react-native-svg` is shared by all SVG components.
