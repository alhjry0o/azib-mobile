import { z } from 'zod';

import { DataOriginSchema, IdSchema, IsoDateSchema } from './common';

export const JobStatusSchema = z.enum(['draft', 'open', 'closed']);

export const JobSchema = z.object({
  id: IdSchema,
  titleKey: z.string().min(1),
  partnerId: IdSchema,
  status: JobStatusSchema,
  createdAt: IsoDateSchema,
  updatedAt: IsoDateSchema,
  dataOrigin: DataOriginSchema,
});

export const CreateJobInputSchema = z.object({
  titleKey: z.string().min(1),
  partnerId: IdSchema,
});

export const UpdateJobInputSchema = z.object({
  titleKey: z.string().min(1).optional(),
  status: JobStatusSchema.optional(),
});
```

import { z } from 'zod';

export const VolunteerStatusSchema = z.enum(['active', 'inactive']);

  skillKey: z.string().min(1),

  createdAt: IsoDateSchema,
  updatedAt: IsoDateSchema,
  dataOrigin: DataOriginSchema,
});

export const CreateVolunteerInputSchema = z.object({
  userId: IdSchema,
  skillKey: z.string().min(1),
});

export const UpdateVolunteerInputSchema = z.object({
  skillKey: z.string().min(1).optional(),
  status: VolunteerStatusSchema.optional(),
});

import { CreateJobInputSchema, UpdateJobInputSchema } from '@/schemas/job';

import type {
  CreateJobInput,
  Job,
  JobRepository,
  UpdateJobInput,
} from '../contracts/job';
import { createMockCrudRepository } from './crud-factory';
import type { MockRepositoryOptions } from './profile';

export function createMockJobRepository(
  options: MockRepositoryOptions = {},
): JobRepository {
  return createMockCrudRepository<Job, CreateJobInput, UpdateJobInput>({
    idPrefix: 'mock-job',
    schemaCreate: CreateJobInputSchema,
    schemaUpdate: UpdateJobInputSchema,
    buildEntity: ({ id, input, nowIso }) => ({
      id,
      titleKey: input.titleKey,
      partnerId: input.partnerId,
      status: 'draft',
      createdAt: nowIso,
      updatedAt: nowIso,
      dataOrigin: 'demo',
    }),
    applyUpdate: ({ existing, patch, nowIso }) => ({
      ...existing,
      ...(patch.titleKey !== undefined && { titleKey: patch.titleKey }),
      ...(patch.status !== undefined && { status: patch.status }),
      updatedAt: nowIso,
    }),
    delayMs: options.delayMs,
    forceErrorKind: options.forceErrorKind ?? null,
  });
}

import { crudRepositoryContract } from '../../__tests__/contracts/crud.contract';
import { createMockJobRepository } from '../job';

describe('MockJobRepository', () => {
  crudRepositoryContract({
    getRepo: () => createMockJobRepository(),
    validCreate: {
      titleKey: 'jobs.demo.title',
      partnerId: 'partner-1',
    },
    invalidCreate: {
      titleKey: '',
      partnerId: '',
    },
    validUpdate: { status: 'open' as const },
    expectedDataOrigin: 'demo',
  });

  it('forceErrorKind: Network causes all methods to fail', async () => {
    const repo = createMockJobRepository({ forceErrorKind: 'Network' });
    const r = await repo.list();
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.kind).toBe('Network');
  });
});
```
