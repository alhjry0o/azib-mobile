// SearchRepository mock delegates to an injected SearchProvider.
// No duplicated search logic. The provider is the single source of the search index.

import { ok, type Result } from '@/core/result';

import type { SearchProvider } from '@/providers/contracts';

import type { SearchRepository } from '../contracts/search';
import type { ListQuery } from '../types';

export function createMockSearchRepository(searchProvider: SearchProvider): SearchRepository {
  return Object.freeze({
    async search<T = Record<string, unknown>>(
      text: string,
      query?: ListQuery,
    ): Promise<Result<readonly T[]>> {
      const r = await searchProvider.query<T>({
        text,
        ...(query?.limit !== undefined && { limit: query.limit }),
      });
      if (!r.ok) return r;
      return ok(r.value.map((hit) => hit.item));
    },
  });
}

import { Pressable, StyleSheet, View } from 'react-native';

import { useHaptics } from '@/hooks';
import { useTheme } from '@/theme';

import { Icon, type IconName } from '../Icon';
import { Text } from '../Text';

export type ChipVariant = 'default' | 'accent' | 'gold';

export interface ChipProps {
  readonly label: string;
  readonly selected?: boolean;
  readonly onPress?: () => void;
  readonly disabled?: boolean;
  readonly leftIcon?: IconName;
  readonly variant?: ChipVariant;
  readonly testID?: string;
}

export function Chip({
  label,
  selected = false,
  onPress,
  disabled = false,
  leftIcon,
  variant = 'default',
  testID,
}: ChipProps) {
  const theme = useTheme();
  const haptics = useHaptics();
  const isInteractive = !!onPress;

  const palette =
    variant === 'accent'
      ? { fill: theme.colors.accent.fill, text: theme.colors.text.primary, border: 'transparent' }
      : variant === 'gold'
        ? { fill: theme.colors.gold.fill, text: theme.colors.text.inverse, border: 'transparent' }
        : {
            fill: selected ? theme.colors.accent.fill : 'transparent',
            text: selected ? theme.colors.text.primary : theme.colors.text.secondary,
            border: selected ? theme.colors.accent.fill : theme.colors.border.subtle,
          };

  const content = (
    <View
      style={[
        styles.container,
        {
          backgroundColor: palette.fill,
          borderColor: palette.border,
          borderRadius: theme.radius.full,
          paddingHorizontal: theme.spacing[3],
          paddingVertical: theme.spacing[1],
          opacity: disabled ? 0.45 : 1,
        },
      ]}
    >
      {leftIcon ? <Icon name={leftIcon} size={14} color={palette.text} /> : null}
      <Text variant="captionStrong" color={palette.text}>
        {label}
      </Text>
    </View>
  );

  if (!isInteractive) return content;

  return (
    <Pressable
      accessibilityRole="button"
      accessibilityState={{ selected, disabled }}
      accessibilityLabel={label}
      onPress={() => {
        if (disabled) return;
        haptics.light();
        onPress?.();
      }}
      disabled={disabled}
      hitSlop={6}
      testID={testID}
    >
      {content}
    </Pressable>
  );
}

const styles = StyleSheet.create({
  container: {
    flexDirection: 'row',
    alignItems: 'center',
    gap: 4,
    borderWidth: 1,
    minHeight: 32,
  },
});

import { forwardRef, useState, type ElementRef } from 'react';

import { TextInput, type TextInputProps } from './TextInput';

export interface SearchProps
  extends Omit<TextInputProps, 'leftIcon' | 'rightIcon' | 'onRightIconPress'> {
  readonly value?: string;
  readonly onClear?: () => void;
}

export const Search = forwardRef<ElementRef<typeof TextInput>, SearchProps>(function Search(
  { value, onClear, onChangeText, ...rest },
  ref,
) {
  const [internalValue, setInternalValue] = useState('');
  const current = value ?? internalValue;
  const showClear = current.length > 0 && !!onClear;

  return (
    <TextInput
      ref={ref}
      {...rest}
      value={value}
      onChangeText={(t) => {
        setInternalValue(t);
        onChangeText?.(t);
      }}
      returnKeyType={rest.returnKeyType ?? 'search'}
      autoCorrect={false}
      autoCapitalize={rest.autoCapitalize ?? 'none'}
      leftIcon="search"
      rightIcon={showClear ? 'x' : undefined}
      onRightIconPress={showClear ? onClear : undefined}
    />
  );
});

import { createMockSearchProvider } from '@/providers/mock';

import { createMockSearchRepository } from '../search';

describe('MockSearchRepository', () => {
  it('delegates to SearchProvider and unwraps items', async () => {
    const provider = createMockSearchProvider(undefined, [
      { id: '1', score: 1, item: { name: 'wedding partner' } },
      { id: '2', score: 1, item: { name: 'souk' } },
    ]);
    const repo = createMockSearchRepository(provider);
    const r = await repo.search<{ name: string }>('wedding');
    expect(r.ok).toBe(true);
    if (r.ok) {
      expect(r.value.length).toBe(1);
      expect(r.value[0]?.name).toBe('wedding partner');
    }
  });

  it('returns empty for empty query', async () => {
    const provider = createMockSearchProvider();
    const repo = createMockSearchRepository(provider);
    const r = await repo.search('   ');
    expect(r.ok).toBe(true);
    if (r.ok) expect(r.value).toEqual([]);
  });

  it('respects limit', async () => {
    const items = Array.from({ length: 10 }, (_, i) => ({
      id: String(i),
      score: 1,
      item: { name: 'match' },
    }));
    const provider = createMockSearchProvider(undefined, items);
    const repo = createMockSearchRepository(provider);
    const r = await repo.search('match', { limit: 3 });
    expect(r.ok).toBe(true);
    if (r.ok) expect(r.value.length).toBe(3);
  });

  it('propagates provider errors', async () => {
    const provider = createMockSearchProvider({ delayMs: 0, forceError: 'Network' });
    const repo = createMockSearchRepository(provider);
    const r = await repo.search('anything');
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.kind).toBe('Network');
  });
});
