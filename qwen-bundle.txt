// Shared contract for AppendOnlyRepository implementations (donation, disbursement).

import type { AppendOnlyRepository, BaseEntity } from '@/repositories/types';

export interface AppendOnlyContractConfig<T extends BaseEntity, TCreate> {
  readonly getRepo: () => AppendOnlyRepository<T, TCreate>;
  readonly validCreate: TCreate;
  readonly invalidCreate: TCreate;
  readonly expectedDataOrigin: 'demo' | 'real';
}

export function appendOnlyRepositoryContract<T extends BaseEntity, TCreate>(
  config: AppendOnlyContractConfig<T, TCreate>,
): void {
  describe('AppendOnlyRepository contract', () => {
    it('getById returns null for unknown id', async () => {
      const repo = config.getRepo();
      const r = await repo.getById('nonexistent');
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('create returns an entity with correct dataOrigin', async () => {
      const repo = config.getRepo();
      const r = await repo.create(config.validCreate);
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.id).toBeTruthy();
        expect(r.value.dataOrigin).toBe(config.expectedDataOrigin);
        expect(r.value.createdAt).toBeTruthy();
        expect(r.value.updatedAt).toBeTruthy();
      }
    });

    it('create rejects invalid input via Validation error', async () => {
      const repo = config.getRepo();
      const r = await repo.create(config.invalidCreate);
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('Validation');
    });

    it('list returns PagedResult with correct totals', async () => {
      const repo = config.getRepo();
      await repo.create(config.validCreate);
      await repo.create(config.validCreate);
      await repo.create(config.validCreate);
      const r = await repo.list();
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.total).toBe(3);
        expect(r.value.items.length).toBe(3);
        expect(r.value.limit).toBe(20);
        expect(r.value.offset).toBe(0);
      }
    });

    it('list respects limit and offset', async () => {
      const repo = config.getRepo();
      for (let i = 0; i < 5; i += 1) {
        await repo.create(config.validCreate);
      }
      const r = await repo.list({ limit: 2, offset: 1 });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.items.length).toBe(2);
        expect(r.value.total).toBe(5);
      }
    });
  });
}

// PaymentProvider contract. Compliant with ADR-0005 (Non-Custodial).
//
// FORBIDDEN in this interface (enforced by absence):
//   - charge(), capture(), refund(), createIntent(), createPaymentMethod()
//   - any method that touches card data, IBAN, or wallet balances.
//
// ALLOWED:
//   - recordPledge(): register an intent to donate (no money moves in-app).
//   - verifyExternalPayment(): verify a reference issued by a licensed external provider (P09).

import type { Result } from '@/core/result';
import type { ProviderBase } from '../types';

export interface DonationPledge {
  readonly amount: number;
  readonly currency: 'YER';
  readonly note?: string;
}

export type PledgeStatus = 'pending' | 'confirmed' | 'declined';

export interface PledgeRecord {
  readonly id: string;
  readonly amount: number;
  readonly currency: 'YER';
  readonly status: PledgeStatus;
  /** ISO 8601. */
  readonly createdAt: string;
  readonly dataOrigin: 'real' | 'demo';
}

export interface PaymentProvider extends ProviderBase {
  readonly name: 'payment';
  recordPledge(pledge: DonationPledge): Promise<Result<PledgeRecord>>;
  verifyExternalPayment(providerReference: string): Promise<Result<PledgeRecord | null>>;
}

// In-memory PaymentProvider. Compliant with ADR-0005 (Non-Custodial).
// No method moves money. recordPledge stores an intent only.
// verifyExternalPayment returns null ALWAYS — no fake confirmation.

import { AppError } from '@/core/errors';
import type { Result } from '@/core/result';

import type { DonationPledge, PaymentProvider, PledgeRecord } from '../contracts/payment';
import type { HealthCheckResult } from '../types';
import { DEFAULT_MOCK_CONFIG, simulateMockIO, stableHash, type MockProviderConfig } from './config';

export interface MockPaymentProvider extends PaymentProvider {
  __pledges(): readonly PledgeRecord[];
}

export function createMockPaymentProvider(
  config: MockProviderConfig = DEFAULT_MOCK_CONFIG,
): MockPaymentProvider {
  const pledges = new Map<string, PledgeRecord>();

  return Object.freeze({
    name: 'payment' as const,
    mode: 'mock' as const,
    capabilities: ['recordPledge', 'verifyExternalPayment'] as const,
    async healthCheck(): Promise<HealthCheckResult> {
      return { status: 'ok', checkedAt: Date.now() };
    },
    async recordPledge(pledge: DonationPledge): Promise<Result<PledgeRecord>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      if (pledge.amount <= 0) {
        return { ok: false, error: AppError.validation({ code: 'INVALID_AMOUNT' }) };
      }
      const id = `mock-pledge-${stableHash(JSON.stringify(pledge) + Date.now())}`;
      const record: PledgeRecord = Object.freeze({
        id,
        amount: pledge.amount,
        currency: 'YER' as const,
        status: 'pending' as const,
        createdAt: new Date().toISOString(),
        dataOrigin: 'demo' as const,
      });
      pledges.set(id, record);
      return { ok: true, value: record };
    },
    async verifyExternalPayment(): Promise<Result<PledgeRecord | null>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      return { ok: true, value: null };
    },
    __pledges(): readonly PledgeRecord[] {
      return Array.from(pledges.values());
    },
  });
}

// Generic in-memory CRUD mock factory.
// One implementation, six repositories. Avoids duplicating ~90 lines per repo.

import type { ZodType } from 'zod';

import { AppError } from '@/core/errors';
import { ok, type Result } from '@/core/result';

import type { BaseEntity, CrudRepository, ListQuery, PagedResult } from '../types';

export interface CrudFactoryOptions<T extends BaseEntity, TCreate, TUpdate> {
  /** Prefix for generated ids. */
  readonly idPrefix: string;
  readonly schemaCreate: ZodType<TCreate>;
  readonly schemaUpdate: ZodType<TUpdate>;
  readonly buildEntity: (args: { id: string; input: TCreate; nowIso: string }) => T;
  readonly applyUpdate: (args: { existing: T; patch: TUpdate; nowIso: string }) => T;
  readonly delayMs?: number;
  readonly forceErrorKind?: 'Network' | 'Unknown' | 'Validation' | null;
}

export function createMockCrudRepository<T extends BaseEntity, TCreate, TUpdate>(
  options: CrudFactoryOptions<T, TCreate, TUpdate>,
): CrudRepository<T, TCreate, TUpdate> {
  const { idPrefix, schemaCreate, schemaUpdate, buildEntity, applyUpdate } = options;
  const delayMs = options.delayMs ?? 0;
  const forceErrorKind = options.forceErrorKind ?? null;
  const byId = new Map<string, T>();
  let counter = 0;

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) {
      return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    }
    return null;
  }

  const repo: CrudRepository<T, TCreate, TUpdate> = {
    async getById(id: string): Promise<Result<T | null>> {
      const failure = await simulate();
      if (failure) return { ok: false, error: failure };
      return ok(byId.get(id) ?? null);
    },

    async list(query?: ListQuery): Promise<Result<PagedResult<T>>> {
      const failure = await simulate();
      if (failure) return { ok: false, error: failure };
      const limit = query?.limit ?? 20;
      const offset = query?.offset ?? 0;
      const all = Array.from(byId.values());
      const items = all.slice(offset, offset + limit);
      return ok({ items, total: all.length, limit, offset });
    },

    async create(input: TCreate): Promise<Result<T>> {
      const parsed = schemaCreate.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const failure = await simulate();
      if (failure) return { ok: false, error: failure };
      counter += 1;
      const id = `${idPrefix}-${Date.now()}-${counter}`;
      const entity = buildEntity({ id, input: parsed.data, nowIso: new Date().toISOString() });
      byId.set(id, entity);
      return ok(entity);
    },

    async update(id: string, patch: TUpdate): Promise<Result<T>> {
      const parsed = schemaUpdate.safeParse(patch);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const failure = await simulate();
      if (failure) return { ok: false, error: failure };
      const existing = byId.get(id);
      if (!existing) {
        return { ok: false, error: AppError.notFound({ code: 'NOT_FOUND' }) };
      }
      const updated = applyUpdate({ existing, patch: parsed.data, nowIso: new Date().toISOString() });
      byId.set(id, updated);
      return ok(updated);
    },

    async delete(id: string): Promise<Result<void>> {
      const failure = await simulate();
      if (failure) return { ok: false, error: failure };
      byId.delete(id);
      return ok(undefined);
    },
  };

  return repo;
}
