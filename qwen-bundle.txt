import type { ApplicationRepository } from '@/repositories/contracts/application';

export function applicationRepositoryContract(getRepo: () => ApplicationRepository): void {
  describe('ApplicationRepository contract', () => {
    it('getById returns null for unknown id', async () => {
      const repo = getRepo();
      const r = await repo.getById('nonexistent');
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('create returns a draft application with dataOrigin=demo', async () => {
      const repo = getRepo();
      const r = await repo.create({
        applicantId: 'a1',
        programId: 'p1',
        requestedAmount: 5000,
      });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.status).toBe('draft');
        expect(r.value.dataOrigin).toBe('demo');
        expect(r.value.currency).toBe('YER');
      }
    });

    it('rejects invalid input via Validation error', async () => {
      const repo = getRepo();
      const r = await repo.create({
        applicantId: '',
        programId: '',
        requestedAmount: -1,
      });
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('Validation');
    });

    it('updateStatus changes status and preserves dataOrigin', async () => {
      const repo = getRepo();
      const created = await repo.create({
        applicantId: 'a2',
        programId: 'p2',
        requestedAmount: 3000,
      });
      if (!created.ok) throw new Error('setup failed');
      const r = await repo.updateStatus(created.value.id, 'reviewing');
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.status).toBe('reviewing');
        expect(r.value.dataOrigin).toBe('demo');
      }
    });

    it('updateStatus returns NotFound for unknown id', async () => {
      const repo = getRepo();
      const r = await repo.updateStatus('nope', 'approved');
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('NotFound');
    });

    it('listByApplicant returns all applications for the applicant', async () => {
      const repo = getRepo();
      await repo.create({ applicantId: 'a3', programId: 'p1', requestedAmount: 1000 });
      await repo.create({ applicantId: 'a3', programId: 'p2', requestedAmount: 2000 });
      await repo.create({ applicantId: 'other', programId: 'p1', requestedAmount: 500 });
      const r = await repo.listByApplicant('a3');
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.length).toBe(2);
        expect(r.value.every((a) => a.applicantId === 'a3')).toBe(true);
      }
    });

    it('list filters by status and programId', async () => {
      const repo = getRepo();
      const c1 = await repo.create({
        applicantId: 'x',
        programId: 'pA',
        requestedAmount: 100,
      });
      await repo.create({ applicantId: 'x', programId: 'pB', requestedAmount: 200 });
      if (!c1.ok) throw new Error('setup failed');
      await repo.updateStatus(c1.value.id, 'approved');
      const r = await repo.list({ status: 'approved' });
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value.total).toBe(1);
    });

    it('delete removes the application', async () => {
      const repo = getRepo();
      const c = await repo.create({ applicantId: 'z', programId: 'p', requestedAmount: 1 });
      if (!c.ok) throw new Error('setup failed');
      const d = await repo.delete(c.value.id);
      expect(d.ok).toBe(true);
      const r = await repo.getById(c.value.id);
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });
  });
}

import { AppError } from '@/core/errors';
import { ok, type Result } from '@/core/result';
import {
  CreateProgramInputSchema,
  UpdateProgramInputSchema,
} from '@/schemas/program';

import type {
  CreateProgramInput,
  Program,
  ProgramRepository,
  UpdateProgramInput,
} from '../contracts/program';
import type { ListQuery, PagedResult } from '../types';
import type { MockRepositoryOptions } from './profile';

export function createMockProgramRepository(
  options: MockRepositoryOptions = {},
): ProgramRepository {
  const { delayMs = 0, forceErrorKind = null } = options;
  const byId = new Map<string, Program>();

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    return null;
  }

  return Object.freeze({
    async getById(id: string): Promise<Result<Program | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      return ok(byId.get(id) ?? null);
    },

    async list(query?: ListQuery): Promise<Result<PagedResult<Program>>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const limit = query?.limit ?? 20;
      const offset = query?.offset ?? 0;
      const all = Array.from(byId.values());
      const items = all.slice(offset, offset + limit);
      return ok({ items, total: all.length, limit, offset });
    },

    async create(input: CreateProgramInput): Promise<Result<Program>> {
      const parsed = CreateProgramInputSchema.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const id = `mock-program-${Date.now()}-${byId.size}`;
      const now = new Date().toISOString();
      const program: Program = Object.freeze({
        id,
        nameKey: parsed.data.nameKey,
        descriptionKey: parsed.data.descriptionKey,
        status: parsed.data.status,
        minAmount: parsed.data.minAmount,
        maxAmount: parsed.data.maxAmount,
        currency: 'YER' as const,
        createdAt: now,
        updatedAt: now,
        dataOrigin: 'demo' as const,
      });
      byId.set(id, program);
      return ok(program);
    },

    async update(id: string, input: UpdateProgramInput): Promise<Result<Program>> {
      const parsed = UpdateProgramInputSchema.safeParse(input);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const existing = byId.get(id);
      if (!existing) {
        return { ok: false, error: AppError.notFound({ code: 'PROGRAM_NOT_FOUND' }) };
      }
      const updated: Program = Object.freeze({
        ...existing,
        ...(parsed.data.nameKey !== undefined && { nameKey: parsed.data.nameKey }),
        ...(parsed.data.descriptionKey !== undefined && {
          descriptionKey: parsed.data.descriptionKey,
        }),
        ...(parsed.data.status !== undefined && { status: parsed.data.status }),
        ...(parsed.data.minAmount !== undefined && { minAmount: parsed.data.minAmount }),
        ...(parsed.data.maxAmount !== undefined && { maxAmount: parsed.data.maxAmount }),
        updatedAt: new Date().toISOString(),
      });
      byId.set(id, updated);
      return ok(updated);
    },

    async delete(id: string): Promise<Result<void>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      byId.delete(id);
      return ok(undefined);
    },
  });
}

// In-memory AiProvider. Compliant with ADR-0006 (Advisory only).
// Returns a deterministic score from input factors, always marked advisory:true.
// Never rejects, approves, or blocks — that is not in the contract.

import type { Result } from '@/core/result';

import type { AiProvider, AiScoreInput, AiScoreOutput } from '../contracts/ai';
import type { HealthCheckResult } from '../types';
import { DEFAULT_MOCK_CONFIG, simulateMockIO, type MockProviderConfig } from './config';

function clamp(n: number, min: number, max: number): number {
  return Math.max(min, Math.min(max, n));
}

export function createMockAiProvider(
  config: MockProviderConfig = DEFAULT_MOCK_CONFIG,
): AiProvider {
  return Object.freeze({
    name: 'ai' as const,
    mode: 'mock' as const,
    capabilities: ['scoreEligibility'] as const,
    async healthCheck(): Promise<HealthCheckResult> {
      return { status: 'ok', checkedAt: Date.now() };
    },
    async scoreEligibility(input: AiScoreInput): Promise<Result<AiScoreOutput>> {
      const failure = await simulateMockIO(config);
      if (failure) return { ok: false, error: failure };
      const values = Object.values(input.factors);
      const avg = values.length > 0 ? values.reduce((a, b) => a + b, 0) / values.length : 0;
      return {
        ok: true,
        value: {
          score: clamp(Math.round(avg), 0, 100),
          advisory: true,
          explanation: 'Deterministic mock score from factors. Advisory only.',
        },
      };
    },
  });
}

# THREAT_MODEL — نموذج تهديد مبدئي (STRIDE)

> مسودة مبدئية. تُراجَع وتُستكمل في P19 (Security) وP18 (Supabase/RLS).
> **عام المستودع:** لا أسرار ولا بيانات شخصية حقيقية، ولا تفاصيل استغلال قابلة للتنفيذ.

---

## 1) الأصول (Assets)

| # | الأصل | الحساسية | سبب الأهمية |
|---|--------|-----------|--------------|
| A1 | بيانات المستفيدين الشخصية (PII) | مرتفعة | قد تكشف وضعاً مالياً/اجتماعياً حساساً. |
| A2 | بيانات المتقدمين للأهلية والمستندات | مرتفعة جداً | قد تحمل مستندات دخل/سكن. |
| A3 | سجلات التبرعات والمساعدات | مرتفعة | سلامة مالية وسمعة. |
| A4 | هويات المتبرعين | متوسطة | خصوصية التبرع. |
| A5 | حسابات المستخدمين وبيانات الاعتماد | مرتفعة | سرقة حساب = تصرف باسم المستخدم. |
| A6 | بيانات الشركاء التجارية | متوسطة | التزامات تعاقدية. |
| A7 | سجلات المراجعة (Audit) | مرتفعة | إثبات عدالة القرار. |
| A8 | الكود المصدري والأسرار التشغيلية | مرتفعة | تسريب مفاتيح = تجاوز كامل. |
| A9 | إحصاءات المنصة العامة | منخفضة | كشفها قد يكشف أفراداً إذا كانت صغيرة. |

---

## 2) المهاجمون (Actors)

| # | المهاجم | الدافع | القدرة |
|---|----------|--------|--------|
| T1 | مهاجم خارجي مجهول | سرقة بيانات / تعطيل | متوسطة |
| T2 | مستخدم مسجَّل يبحث عن تجاوز صلاحيات | كشف بيانات مستفيدين | منخفضة–متوسطة |
| T3 | متقدم محتال (تقديم بيانات كاذبة) | الحصول على مساعدة | متوسطة |
| T4 | شريك/سفير غير أمين | كشف هوية المستفيدين | متوسطة |
| T5 | موظف داخلي (Insider) | سوء استخدام صلاحيات | مرتفعة (وصول مباشر) |
| T6 | مزوّد خدمة خارجي مخترق | كشف بيانات | متوسطة |
| T7 | مهاجم يستهدف CI/سلسلة التوريد | حقن كود/سرقة أسرار | متوسطة |

---

## 3) أسطح الهجوم (Attack Surfaces)

| # | السطح | أمثلة |
|---|--------|--------|
| S1 | تطبيق الموبايل | تخزين محلي غير مشفّر، Deeplinks، شاشات مسربة. |
| S2 | واجهة Supabase | RLS غير محكم، استعلامات مفتوحة. |
| S3 | المزوّد المجرّد | تسريب مفاتيح، استدعاءات غير مصرّح بها. |
| S4 | التخزين (Storage) | ملفات عامة، مسارات قابلة للتخمين. |
| S5 | المصادقة | كلمات ضعيفة، جلسات طويلة، OTP بدون حد. |
| S6 | البحث والإحصاءات | كشف أفراد عبر تجميعات صغيرة. |
| S7 | CI/CD | أسرار في المستودع، صلاحيات Actions واسعة. |
| S8 | Sideloading/APK معدّل | تجاوز فحوص العميل. |
| S9 | المزوّد المرخّص للتبرعات | Webhook بدون توقيع. |

---

## 4) تحليل STRIDE

### S — Spoofing (انتحال)
- **تهديد:** T3 ينتحل هوية متقدم آخر، أو T5 ينتحل دور أعلى.
- **ضوابط مبدئية:** مصادقة قوية، جلسات محدودة العمر، فصل أدوار صريح، فحص دور في كل مسار ولوحة إدارية، MFA إلزامي للموظفين في P19.

### T — Tampering (تعديل)
- **تهديد:** تعديل مستندات أهلية أو مبالغ مساعدة.
- **ضوابط:** سجل مراجعة Append-only، توقيع/بصمة على المستندات، منع تعديل السجلات المالية من الواجهة، RLS دقيق في P18.

### R — Repudiation (إنكار)
- **تهديد:** إنكار مراجع قرار رفض/قبول.
- **ضوابط:** تسجيل كل قرار مع هوية المراجع ووقته والسبب، وإتاحة سجلات للمدقق (auditor).

### I — Information Disclosure (كشف معلومات)
- **تهديد:** كشف PII لشريك/سفير/مستخدم عادي، كشف أفراد عبر إحصاءات صغيرة.
- **ضوابط:** RLS، إخفاء الإحصاءات < الحد الأدنى، لا كشف هوية المستفيد للشريك، عرض «الحد الأدنى من الظهور»، تشفير نقل و(عند الحاجة) تشفير تخزين على مستوى الأعمدة الحساسة.

### D — Denial of Service
- **تهديد:** إغراق طلبات التقديم/التبرع.
- **ضوابط:** Rate limiting، فحص Human (عند الحاجة)، طوابير للعمليات الثقيلة.

### E — Elevation of Privilege
- **تهديد:** عضو عادي يصل لبيانات مستفيدين، Insider يتجاوز المراجعة.
- **ضوابط:** فصل الأدوار، مبدأ أقل صلاحية، مراجعة دورية للصلاحيات، سجلات تدقيق، فحوص RLS في pgTAP.

---

## 5) ضوابط عامة إلزامية

1. لا أسرار في التطبيق، ولا `service_role` في الكود المخزّن. كل الأسرار في GitHub Secrets أو `.env.local` خارج الـ repo.
2. لا بيانات بطاقات، لا بنوك داخل التطبيق — لا PCI scope.
3. لا عملية مالية داخلية؛ التبرعات الحقيقية عبر مزوّد مرخّص فقط (ADR-0005).
4. `dataOrigin='demo'` إلزامي في كل Seed، ووسم Demo مرئي للمستخدم.
5. كل مسار إداري محمي بدور صريح + RLS + سجل تدقيق.
6. لا كشف لبيانات المستفيدين لغير المخوّلين (AGENT_RULES #20).
7. المتغيرات البيئية الحساسة تُوثَّق في `PROJECT_CONTEXT.md` (اسم فقط، لا قيمة).

/**
 * Logger — تسجيل آمن بلا PII.
 *
 * ضوابط إخفاء الهوية:
 * - بريد إلكتروني → [REDACTED_EMAIL]
 * - رقم هاتف (يمني أو عام) → [REDACTED_PHONE]
 * - توكن Bearer / JWT → [REDACTED_TOKEN]
 * - أرقام بطاقة (13–19 رقماً) → [REDACTED_CARD]
 *
 * القاعدة: لا تسجّل أبداً كائنات كاملة تحتوي بيانات مستخدم.
 * مرّر حقولاً منقّاة أو استخدم redact() صراحةً.
 */

export type LogLevel = 'debug' | 'info' | 'warn' | 'error';

const REDACT_PATTERNS: readonly (readonly [RegExp, string])[] = [
  // JWT (three base64url segments starting with eyJ)
  [/eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+/g, '[REDACTED_TOKEN]'],
  // Bearer tokens
  [/(bearer\s+)[A-Za-z0-9._-]+/gi, '$1[REDACTED_TOKEN]'],
  // Email
  [/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '[REDACTED_EMAIL]'],
  // Card-like (13–19 digits, optionally grouped)
  [/\b(?:\d[ -]*?){13,19}\b/g, '[REDACTED_CARD]'],
  // Phone (Yemen + generic + E.164)
  [/(?:\+?\d{1,3}[-\s]?)?(?:0|\d)?\d{7,12}/g, '[REDACTED_PHONE]'],
];

export function redactString(input: string): string {
  let out = input;
  for (const [pattern, replacement] of REDACT_PATTERNS) {
    out = out.replace(pattern, replacement);
  }
  return out;
}

export function redact(value: unknown, depth = 0): unknown {
  if (depth > 6) return '[DEPTH_LIMIT]';
  if (value === null || value === undefined) return value;
  if (typeof value === 'string') return redactString(value);
  if (typeof value === 'number' || typeof value === 'boolean') return value;
  if (value instanceof Error) return { name: value.name, message: redactString(value.message) };
  if (Array.isArray(value)) return value.map((v) => redact(v, depth + 1));
  if (typeof value === 'object') {
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
      out[k] = redact(v, depth + 1);
    }
    return out;
  }
  return '[UNKNOWN]';
}

export interface Logger {
  debug(message: string, context?: Record<string, unknown>): void;
  info(message: string, context?: Record<string, unknown>): void;
  warn(message: string, context?: Record<string, unknown>): void;
  error(message: string, context?: Record<string, unknown>): void;
}

const isDev = (): boolean => {
  const proc = (globalThis as any).process;
  return proc?.env?.NODE_ENV !== 'production';
};

export const logger: Logger = {
  debug(message, context) {
    if (!isDev()) return;
    console.debug('[debug]', redactString(message), context ? redact(context) : undefined);
  },
  info(message, context) {
    if (!isDev()) return;
    console.info('[info]', redactString(message), context ? redact(context) : undefined);
  },
  warn(message, context) {
    console.warn('[warn]', redactString(message), context ? redact(context) : undefined);
  },
  error(message, context) {
    console.error('[error]', redactString(message), context ? redact(context) : undefined);
  },
};
