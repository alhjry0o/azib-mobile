// Generic CRUD contract test. Shared across all CrudRepository implementations.
// Each repo wrapper provides: valid create, invalid create, valid update, expected dataOrigin.

import type { BaseEntity, CrudRepository } from '@/repositories/types';

export interface CrudContractConfig<T extends BaseEntity, TCreate, TUpdate> {
  readonly getRepo: () => CrudRepository<T, TCreate, TUpdate>;
  readonly validCreate: TCreate;
  readonly invalidCreate: TCreate;
  readonly validUpdate: TUpdate;
  readonly expectedDataOrigin: 'demo' | 'real';
}

export function crudRepositoryContract<T extends BaseEntity, TCreate, TUpdate>(
  config: CrudContractConfig<T, TCreate, TUpdate>,
): void {
  describe('CrudRepository contract', () => {
    it('getById returns null for unknown id', async () => {
      const repo = config.getRepo();
      const r = await repo.getById('nonexistent');
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('create returns an entity with correct dataOrigin and timestamps', async () => {
      const repo = config.getRepo();
      const r = await repo.create(config.validCreate);
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.id).toBeTruthy();
        expect(r.value.dataOrigin).toBe(config.expectedDataOrigin);
        expect(r.value.createdAt).toBeTruthy();
        expect(r.value.updatedAt).toBeTruthy();
      }
    });

    it('rejects invalid input via Validation error', async () => {
      const repo = config.getRepo();
      const r = await repo.create(config.invalidCreate);
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('Validation');
    });

    it('created entity can be retrieved by id', async () => {
      const repo = config.getRepo();
      const created = await repo.create(config.validCreate);
      if (!created.ok) throw new Error('setup failed');
      const r = await repo.getById(created.value.id);
      expect(r.ok).toBe(true);
      if (r.ok && r.value) expect(r.value.id).toBe(created.value.id);
    });

    it('update modifies fields and updatedAt', async () => {
      const repo = config.getRepo();
      const created = await repo.create(config.validCreate);
      if (!created.ok) throw new Error('setup failed');
      const r = await repo.update(created.value.id, config.validUpdate);
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.updatedAt >= created.value.updatedAt).toBe(true);
        expect(r.value.dataOrigin).toBe(config.expectedDataOrigin);
      }
    });

    it('update returns NotFound for unknown id', async () => {
      const repo = config.getRepo();
      const r = await repo.update('nonexistent', config.validUpdate);
      expect(r.ok).toBe(false);
      if (!r.ok) expect(r.error.kind).toBe('NotFound');
    });

    it('delete removes the entity', async () => {
      const repo = config.getRepo();
      const created = await repo.create(config.validCreate);
      if (!created.ok) throw new Error('setup failed');
      const d = await repo.delete(created.value.id);
      expect(d.ok).toBe(true);
      const r = await repo.getById(created.value.id);
      expect(r.ok).toBe(true);
      if (r.ok) expect(r.value).toBe(null);
    });

    it('list returns PagedResult with correct totals', async () => {
      const repo = config.getRepo();
      await repo.create(config.validCreate);
      await repo.create(config.validCreate);
      await repo.create(config.validCreate);
      const r = await repo.list();
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.total).toBe(3);
        expect(r.value.items.length).toBe(3);
        expect(r.value.limit).toBe(20);
        expect(r.value.offset).toBe(0);
      }
    });

    it('list respects limit and offset', async () => {
      const repo = config.getRepo();
      for (let i = 0; i < 5; i += 1) {
        await repo.create(config.validCreate);
      }
      const r = await repo.list({ limit: 2, offset: 1 });
      expect(r.ok).toBe(true);
      if (r.ok) {
        expect(r.value.items.length).toBe(2);
        expect(r.value.total).toBe(5);
        expect(r.value.limit).toBe(2);
        expect(r.value.offset).toBe(1);
      }
    });
  });
}

import { AppError } from '@/core/errors';
import { ok, type Result } from '@/core/result';

import type { InAppNotification, NotificationRepository } from '../contracts/notification';
import type { ListQuery, PagedResult } from '../types';
import type { MockRepositoryOptions } from './profile';

export function createMockNotificationRepository(
  options: MockRepositoryOptions = {},
  seed: readonly InAppNotification[] = [],
): NotificationRepository {
  const { delayMs = 0, forceErrorKind = null } = options;
  const byId = new Map<string, InAppNotification>();
  for (const record of seed) byId.set(record.id, record);

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    return null;
  }

  return Object.freeze({
    async getById(id: string): Promise<Result<InAppNotification | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      return ok(byId.get(id) ?? null);
    },

    async list(query?: ListQuery): Promise<Result<PagedResult<InAppNotification>>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const limit = query?.limit ?? 20;
      const offset = query?.offset ?? 0;
      const all = Array.from(byId.values());
      const items = all.slice(offset, offset + limit);
      return ok({ items, total: all.length, limit, offset });
    },

    async markRead(id: string): Promise<Result<InAppNotification>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const existing = byId.get(id);
      if (!existing) {
        return { ok: false, error: AppError.notFound({ code: 'NOTIFICATION_NOT_FOUND' }) };
      }
      const updated: InAppNotification = Object.freeze({
        ...existing,
        read: true,
        updatedAt: new Date().toISOString(),
      });
      byId.set(id, updated);
      return ok(updated);
    },

    async delete(id: string): Promise<Result<void>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      byId.delete(id);
      return ok(undefined);
    },
  });
}

import { AppError } from '@/core/errors';
import { ok, type Result } from '@/core/result';
import { SettingsValueSchema } from '@/schemas/settings';

import type { Settings, SettingsRepository } from '../contracts/settings';
import type { MockRepositoryOptions } from './profile';

export function createMockSettingsRepository(
  options: MockRepositoryOptions = {},
): SettingsRepository {
  const { delayMs = 0, forceErrorKind = null } = options;
  const map = new Map<string, Settings>();

  async function simulate(): Promise<AppError | null> {
    if (delayMs > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, delayMs));
    }
    if (forceErrorKind) return AppError.fromKind(forceErrorKind, { code: 'MOCK_FORCED_ERROR' });
    return null;
  }

  return Object.freeze({
    async get(key: string): Promise<Result<Settings | null>> {
      const f = await simulate();
      if (f) return { ok: false, error: f };
      return ok(map.get(key) ?? null);
    },

    async update(
      key: string,
      value: Readonly<Record<string, unknown>>,
    ): Promise<Result<Settings>> {
      const parsed = SettingsValueSchema.safeParse(value);
      if (!parsed.success) {
        return { ok: false, error: AppError.validation({ cause: parsed.error }) };
      }
      const f = await simulate();
      if (f) return { ok: false, error: f };
      const record: Settings = Object.freeze({
        key,
        value: Object.freeze(parsed.data),
        updatedAt: new Date().toISOString(),
      });
      map.set(key, record);
      return ok(record);
    },
  });
}

import { useState } from 'react';
import { Pressable, StyleSheet } from 'react-native';
import { useTranslation } from 'react-i18next';

import { formatDate } from '@/i18n';
import { useHaptics } from '@/hooks';
import { useTheme } from '@/theme';

import { Icon } from '../Icon';
import { Text } from '../Text';
import { CalendarModal } from './CalendarModal';
import { Field } from './Field';

export interface DatePickerProps {
  readonly value: Date | null;
  readonly onChange: (date: Date | null) => void;
  readonly minDate?: Date;
  readonly maxDate?: Date;
  readonly placeholder?: string;
  readonly label?: string;
  readonly required?: boolean;
  readonly helpText?: string;
  readonly errorText?: string;
  readonly error?: boolean;
  readonly disabled?: boolean;
  readonly testID?: string;
}

export function DatePicker({
  value,
  onChange,
  minDate,
  maxDate,
  placeholder,
  label,
  required = false,
  helpText,
  errorText,
  error = false,
  disabled = false,
  testID,
}: DatePickerProps) {
  const theme = useTheme();
  const { i18n } = useTranslation();
  const haptics = useHaptics();
  const [open, setOpen] = useState(false);
  const hasError = error || !!errorText;

  const language = i18n.language.startsWith('ar') ? 'ar' : 'en';
  const display = value ? formatDate(value, 'medium', { language }) : '';

  const triggerBorder = hasError
    ? theme.colors.state.danger
    : open
      ? theme.colors.accent.fill
      : theme.colors.border.subtle;

  return (
    <Field
      label={label}
      required={required}
      helpText={helpText}
      errorText={errorText}
      disabled={disabled}
      testID={testID ? `${testID}-field` : undefined}
    >
      <Pressable
        accessibilityRole="button"
        accessibilityState={{ disabled, expanded: open }}
        accessibilityLabel={display || placeholder}
        disabled={disabled}
        onPress={() => {
          if (disabled) return;
          haptics.light();
          setOpen(true);
        }}
        testID={testID}
        style={[
          styles.trigger,
          {
            backgroundColor: theme.colors.bg.surface,
            borderColor: triggerBorder,
            borderWidth: open || hasError ? 2 : 1,
            borderRadius: theme.radius.md,
            paddingStart: theme.spacing[3],
            paddingEnd: theme.spacing[3],
            opacity: disabled ? 0.5 : 1,
          },
        ]}
      >
        <Text variant="body" color={display ? 'primary' : 'tertiary'}>
          {display || placeholder || ''}
        </Text>
        <Icon name="calendar" size={18} color={theme.colors.text.secondary} />
      </Pressable>

      <CalendarModal
        visible={open}
        value={value}
        minDate={minDate}
        maxDate={maxDate}
        onClose={() => setOpen(false)}
        onSelect={(date) => {
          onChange(date);
          setOpen(false);
        }}
      />
    </Field>
  );
}

const styles = StyleSheet.create({
  trigger: {
    minHeight: 48,
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'space-between',
    gap: 8,
  },
});

import { Pressable, StyleSheet, View } from 'react-native';

import { useHaptics } from '@/hooks';
import { useTheme } from '@/theme';

import { Text } from '../Text';

export interface RadioOption<V extends string> {
  readonly value: V;
  readonly label: string;
  readonly disabled?: boolean;
}

export interface RadioGroupProps<V extends string> {
  readonly options: readonly RadioOption<V>[];
  readonly value: V | null;
  readonly onValueChange: (value: V) => void;
  readonly disabled?: boolean;
  readonly direction?: 'row' | 'column';
  readonly error?: boolean;
  readonly testID?: string;
}

export function RadioGroup<V extends string>({
  options,
  value,
  onValueChange,
  disabled = false,
  direction = 'column',
  error = false,
  testID,
}: RadioGroupProps<V>) {
  const theme = useTheme();
  const haptics = useHaptics();

  return (
    <View
      accessibilityRole="radiogroup"
      style={[direction === 'row' ? styles.row : styles.column, { gap: theme.spacing[3] }]}
      testID={testID}
    >
      {options.map((opt) => {
        const selected = value === opt.value;
        const isDisabled = disabled || !!opt.disabled;
        const borderColor = error
          ? theme.colors.state.danger
          : selected
            ? theme.colors.accent.fill
            : theme.colors.border.strong;

        return (
          <Pressable
            key={opt.value}
            accessibilityRole="radio"
            accessibilityState={{ selected, disabled: isDisabled }}
            accessibilityLabel={opt.label}
            onPress={() => {
              if (isDisabled) return;
              haptics.light();
              onValueChange(opt.value);
            }}
            disabled={isDisabled}
            hitSlop={8}
            style={[styles.option, { opacity: isDisabled ? 0.5 : 1 }]}
          >
            <View style={[styles.outer, { borderColor }]}>
              {selected ? (
                <View style={[styles.inner, { backgroundColor: theme.colors.accent.fill }]} />
              ) : null}
            </View>
            <Text variant="body" color={isDisabled ? 'tertiary' : 'primary'}>
              {opt.label}
            </Text>
          </Pressable>
        );
      })}
    </View>
  );
}
