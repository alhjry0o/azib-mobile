import { z } from 'zod';

import { DataOriginSchema, IdSchema, IsoDateSchema } from './common';

export const SoukStatusSchema = z.enum(['available', 'reserved', 'claimed']);

export const SoukItemSchema = z.object({
  id: IdSchema,
  titleKey: z.string().min(1),
  category: z.string().min(1).max(100),
  status: SoukStatusSchema,
  createdAt: IsoDateSchema,
  updatedAt: IsoDateSchema,
  dataOrigin: DataOriginSchema,
});

export const CreateSoukItemInputSchema = z.object({
  titleKey: z.string().min(1),
  category: z.string().min(1).max(100),
});

export const UpdateSoukItemInputSchema = z.object({
  titleKey: z.string().min(1).optional(),
  category: z.string().min(1).max(100).optional(),
  status: SoukStatusSchema.optional(),
});

import { CreateSoukItemInputSchema, UpdateSoukItemInputSchema } from '@/schemas/souk';

import type {
  CreateSoukItemInput,
  SoukItem,
  SoukRepository,
  UpdateSoukItemInput,
} from '../contracts/souk';
import { createMockCrudRepository } from './crud-factory';
import type { MockRepositoryOptions } from './profile';

export function createMockSoukRepository(
  options: MockRepositoryOptions = {},
): SoukRepository {
  return createMockCrudRepository<SoukItem, CreateSoukItemInput, UpdateSoukItemInput>({
    idPrefix: 'mock-souk',
    schemaCreate: CreateSoukItemInputSchema,
    schemaUpdate: UpdateSoukItemInputSchema,
    buildEntity: ({ id, input, nowIso }) => ({
      id,
      titleKey: input.titleKey,
      category: input.category,
      status: 'available',
      createdAt: nowIso,
      updatedAt: nowIso,
      dataOrigin: 'demo',
    }),
    applyUpdate: ({ existing, patch, nowIso }) => ({
      ...existing,
      ...(patch.titleKey !== undefined && { titleKey: patch.titleKey }),
      ...(patch.category !== undefined && { category: patch.category }),
      ...(patch.status !== undefined && { status: patch.status }),
      updatedAt: nowIso,
    }),
    delayMs: options.delayMs,
    forceErrorKind: options.forceErrorKind ?? null,
  });
}

import { crudRepositoryContract } from '../../__tests__/contracts/crud.contract';
import { createMockSoukRepository } from '../souk';

describe('MockSoukRepository', () => {
  crudRepositoryContract({
    getRepo: () => createMockSoukRepository(),
    validCreate: {
      titleKey: 'souk.demo.title',
      category: 'furniture',
    },
    invalidCreate: {
      titleKey: '',
      category: '',
    },
    validUpdate: { status: 'reserved' as const },
    expectedDataOrigin: 'demo',
  });

  it('forceErrorKind: Network causes all methods to fail', async () => {
    const repo = createMockSoukRepository({ forceErrorKind: 'Network' });
    const r = await repo.list();
    expect(r.ok).toBe(false);
    if (!r.ok) expect(r.error.kind).toBe('Network');
  });
});

import type { Result } from '@/core/result';

import type { BaseEntity, ListQuery, PagedResult } from '../types';

export type ProfileGender = 'male' | 'female' | 'unspecified';
export type MaritalStatus = 'single' | 'engaged' | 'married' | 'divorced' | 'widowed';

export interface Profile extends BaseEntity {
  readonly userId: string;
  readonly displayName: string;
  readonly gender: ProfileGender;
  readonly maritalStatus: MaritalStatus;
  readonly city: string | null;
  /** PII. Never expose publicly (AGENT_RULES §1 #20). */
  readonly phone: string | null;
}

export interface CreateProfileInput {
  readonly userId: string;
  readonly displayName: string;
  readonly gender: ProfileGender;
  readonly maritalStatus: MaritalStatus;
  readonly city?: string | null;
  readonly phone?: string | null;
}

export interface UpdateProfileInput {
  readonly displayName?: string;
  readonly gender?: ProfileGender;
  readonly maritalStatus?: MaritalStatus;
  readonly city?: string | null;
  readonly phone?: string | null;
}

export interface ProfileListQuery extends ListQuery {
  readonly city?: string;
  readonly maritalStatus?: MaritalStatus;
}

export interface ProfileRepository {
  getById(id: string): Promise<Result<Profile | null>>;
  getByUserId(userId: string): Promise<Result<Profile | null>>;
  create(input: CreateProfileInput): Promise<Result<Profile>>;
  update(id: string, input: UpdateProfileInput): Promise<Result<Profile>>;
  delete(id: string): Promise<Result<void>>;
  list(query?: ProfileListQuery): Promise<Result<PagedResult<Profile>>>;
}
